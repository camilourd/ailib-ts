function numericGradient(f: (tensor: Tensor) => Tensor, tensor: Tensor, epsilon = 1e-2): Float32Array {
    const original = tensor.data.slice();
    const result = new Float32Array(tensor.size);
    for (let index = 0; index < tensor.size; index++) {
        tensor.data[index] = (original[index] as number) + epsilon;
        const plus = f(tensor).get();
        tensor.data[index] = (original[index] as number) - epsilon;
        const minus = f(tensor).get();
        tensor.data[index] = original[index] as number;
        result[index] = (plus - minus) / (2 * epsilon);
    }
    return result;
}

test("mse: valor y gradiente", () => {
    const prediction = Tensor.from([1, 2], [2], { requiresGrad: true });
    const target = Tensor.from([0, 0], [2], { requiresGrad: true });

    assert.equal(loss.get(), 2.5);
    loss.backward();
    assert.deepEqual([...(prediction.grad?.data ?? [])], [1, 2]);
    assert.deepEqual([...(target.grad?.data ?? [])], [-1, -2]);
});

    let previous = -Infinity;
    for (const x of POINTS) {
        const value = sigmoid(x);
        assert.ok(value > 0 && value < 1, `σ(${x}) = ${value} fuera de (0,1)`);
        assert.ok(value > previous, `σ no es creciente en x = ${x}`);
        previous = value;
    }

test("simetría: σ(-x) = 1 - σ(x)", () => {
    for (const x of POINTS) {
        assert.ok(Math.abs(sigmoid(-x) + sigmoid(x) - 1) < 1e-12, `falla en x = ${x}`);
    }

    const h = 1e-6;
    for (const x of POINTS) {
        const numeric = (sigmoid(x + h) - sigmoid(x - h)) / (2 * h);
        const analytic = sigmoidDerivative(x);
        assert.ok(
            Math.abs(numeric - analytic) < 1e-8,
            `en x = ${x}: numérica ${numeric}, analítica ${analytic}`,

    }
});

    for (const x of [-50, 50]) {

test("SwiGLU: gradiente numérico de la entrada", () => {
    const layer = new SwiGLU(2, 3, { seed: 2 });
    const input = Tensor.from([0.5, -1, 1.5, 2], [2, 2], { requiresGrad: true });
    const loss = () => sum(layer.forward(input)).get();
    sum(layer.forward(input)).backward();
    for (let index = 0; index < input.size; index++) {
        const original = input.data[index] as number;
        input.data[index] = original + 1e-2;
        const plus = loss();
        input.data[index] = original - 1e-2;
        const minus = loss();
        input.data[index] = original;
        const numeric = (plus - minus) / 0.02;
        const analytic = input.grad?.data[index] ?? 0;
        assert.ok(Math.abs(analytic - numeric) < 1e-2, `${index}: ${analytic} vs ${numeric}`);
    }
});

test("entrena XOR con una capa oculta", () => {
    const inputs = Tensor.from([0, 0, 0, 1, 1, 0, 1, 1], [4, 2]);
    const targets = [0, 1, 1, 0];
    const hidden = new Linear(2, 4, { seed: 1 });
    const output = new Linear(4, 2, { seed: 2 });
    const optimizer = new SGD([...hidden.parameters(), ...output.parameters()], 0.5);

    const history = train(
        (input) => output.forward(relu(hidden.forward(input))),
        inputs,
        {
            steps: 3000,
            optimizer,
            loss: (prediction) => crossEntropy(prediction, targets),
        },
    );

    assert.equal(history.length, 3000);
    const finalLoss = history.at(-1) as number;
    assert.ok(finalLoss < 0.02, `pérdida final ${finalLoss}`);

    const logits = output.forward(relu(hidden.forward(inputs)));
    let correct = 0;
    for (let row = 0; row < 4; row++) {
        const predicted = logits.get(row, 1) > logits.get(row, 0) ? 1 : 0;
        if (predicted === targets[row]) correct += 1;
    }
    assert.equal(correct, 4);
});

test("train valida pasos y pérdida escalar", () => {
    const optimizer = new SGD([], 0.1);
    const input = Tensor.from([1], [1]);
    assert.deepEqual(train((value) => value, input, { steps: 0, optimizer, loss: (p) => p }), []);
    assert.throws(
        () => train((value) => value, input, { steps: 1, optimizer, loss: (p) => p }),
        Error,
    );
});

test("onStep recibe el paso y la pérdida", () => {
    const parameter = Tensor.from([1], [1], { requiresGrad: true });
    const optimizer = new SGD([parameter], 0.1);
    const seen: Array<[number, number]> = [];
    const history = train(
        (value) => value,
        Tensor.from([0], [1]),
        {
            steps: 2,
            optimizer,
            loss: () => mean(mul(parameter, parameter)),
            onStep: (step, loss) => seen.push([step, loss]),
        },
    );
    assert.equal(history.length, 2);
    assert.deepEqual(seen.map(([step]) => step), [0, 1]);
    assert.ok((history[0] as number) > (history[1] as number));
});
