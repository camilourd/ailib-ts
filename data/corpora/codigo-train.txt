/** Union-Find con compresión de caminos y unión por tamaño. */
export class DisjointSetUnion {
    #parents: Int32Array;
    #sizes: Int32Array;

    constructor(size = 0) {
        this.#parents = new Int32Array(0);
        this.#sizes = new Int32Array(0);
        this.restart(size);
    }

    /** Reinicia la estructura con `size` elementos aislados. */
    restart(size: number): void {
        if (!Number.isInteger(size) || size < 0) {
            throw new Error(`tamaño inválido: ${size}`);
        }
        this.#parents = new Int32Array(size);
        this.#sizes = new Int32Array(size);
        for (let index = 0; index < size; index++) {
            this.#parents[index] = index;
            this.#sizes[index] = 1;
        }
    }

    get size(): number {
        return this.#parents.length;
    }

    find(value: number): number {
        this.#check(value);
        let root = value;
        while ((this.#parents[root] as number) !== root) {
            root = this.#parents[root] as number;
        }
        let current = value;
        while ((this.#parents[current] as number) !== current) {
            const next = this.#parents[current] as number;
            this.#parents[current] = root;
            current = next;
        }
        return root;
    }

    join(first: number, second: number): void {
        let rootFirst = this.find(first);
        let rootSecond = this.find(second);
        if (rootFirst === rootSecond) return;
        if ((this.#sizes[rootFirst] as number) < (this.#sizes[rootSecond] as number)) {
            const swap = rootFirst;
            rootFirst = rootSecond;
            rootSecond = swap;
        }
        this.#parents[rootSecond] = rootFirst;
        this.#sizes[rootFirst] =
            (this.#sizes[rootFirst] as number) + (this.#sizes[rootSecond] as number);
    }

    #check(value: number): void {
        if (!Number.isInteger(value) || value < 0 || value >= this.size) {
            throw new RangeError(`índice ${value} fuera de [0, ${this.size})`);
        }
    }
}

/** mulberry32: estado de 32 bits y periodo 2^32. */
export class Mulberry32 extends Uint32RandomGenerator {
    #state: number;

    constructor(seed: number) {
        super();
        this.#state = seed >>> 0;
    }

    protected nextUint32(): number {
        this.#state = (this.#state + 0x6d2b79f5) >>> 0;
        let t = Math.imul(this.#state ^ (this.#state >>> 15), this.#state | 1);
        t = (t ^ (t + Math.imul(t ^ (t >>> 7), t | 61))) >>> 0;
        return (t ^ (t >>> 14)) >>> 0;
    }
}

/** Contrato que usa el resto de la librería: no depende del algoritmo. */
export interface RandomNumberGenerator {
    /** Siguiente número uniforme en [0, 1). */
    next(): number;
    /** Entero uniforme en [0, maxExclusive). */
    int(maxExclusive: number): number;
    /** Copia mezclada de `values` (Fisher-Yates); no muta la entrada. */
    shuffle<T>(values: readonly T[]): T[];
}

const TWO_POW_32 = 0x100000000;

/**
 * Base de los generadores de 32 bits (Template Method): `next`, `int` y `shuffle` se implementan
 * una sola vez aquí; cada algoritmo solo aporta su paso crudo en `nextUint32`.
 */
export abstract class Uint32RandomGenerator implements RandomNumberGenerator {
    protected abstract nextUint32(): number;

    next(): number {
        return this.nextUint32() / TWO_POW_32;
    }

    int(maxExclusive: number): number {
        if (!Number.isInteger(maxExclusive) || maxExclusive <= 0 || maxExclusive > TWO_POW_32) {
            throw new RangeError(`maxExclusive debe ser un entero en (0, 2^32], no ${maxExclusive}`);
        }
        const limit = TWO_POW_32 - (TWO_POW_32 % maxExclusive);
        let value = this.nextUint32();
        while (value >= limit) value = this.nextUint32();
        return value % maxExclusive;
    }

    shuffle<T>(values: readonly T[]): T[] {
        const copy = [...values];
        // i y j están siempre dentro del rango por construcción; el cast solo satisface
        // noUncheckedIndexedAccess.
        for (let i = copy.length - 1; i > 0; i--) {
            const j = this.int(i + 1);
            const swap = copy[i] as T;
            copy[i] = copy[j] as T;
            copy[j] = swap;
        }
        return copy;
    }
}

import { Tensor } from "./tensor.ts";

/** Forma resultante de combinar dos formas con las reglas de broadcasting. */
export function broadcastShapes(left: readonly number[], right: readonly number[]): number[] {
    const rank = Math.max(left.length, right.length);
    const shape = new Array<number>(rank);
    for (let axis = 0; axis < rank; axis++) {
        const leftDimension = left[left.length - 1 - axis] ?? 1;
        const rightDimension = right[right.length - 1 - axis] ?? 1;
        if (leftDimension === rightDimension) {
            shape[rank - 1 - axis] = leftDimension;
        } else if (leftDimension === 1) {
            shape[rank - 1 - axis] = rightDimension;
        } else if (rightDimension === 1) {
            shape[rank - 1 - axis] = leftDimension;
        } else {
            throw new Error(
                `formas incompatibles para broadcasting: ${JSON.stringify(left)} y ${JSON.stringify(right)}`,
            );
        }
    }
    return shape;
}

/** Strides del operando `shape` vistos desde `target`; 0 en las dimensiones expandidas. */
export function broadcastStrides(
    shape: readonly number[],
    strides: readonly number[],
    target: readonly number[],
): number[] {
    const result = new Array<number>(target.length).fill(0);
    for (let axis = 0; axis < target.length; axis++) {
        const sourceAxis = shape.length - (target.length - axis);
        if (sourceAxis >= 0) {
            const dimension = shape[sourceAxis] as number;
            if (dimension === target[axis]) {
                result[axis] = strides[sourceAxis] as number;
            } else if (dimension !== 1) {
                throw new Error(`la dimensión ${sourceAxis} (${dimension}) no encaja en ${target[axis]}`);
            }
        }
    }
    return result;
}

/** Suma el gradiente sobre las dimensiones que el operando había expandido. */
function unbroadcast(gradient: Tensor, shape: readonly number[]): Tensor {
    const sameShape =
        gradient.shape.length === shape.length &&
        gradient.shape.every((dimension, axis) => dimension === shape[axis]);
    if (sameShape) return gradient;
    const result = Tensor.zeros(shape);
    for (let linear = 0; linear < gradient.size; linear++) {
        let remainder = linear;
        let offset = 0;
        for (let axis = 0; axis < gradient.rank; axis++) {
            const stride = gradient.strides[axis] as number;
            const index = Math.floor(remainder / stride);
            remainder -= index * stride;
            const targetAxis = shape.length - (gradient.rank - axis);
            if (targetAxis >= 0) {
                const targetIndex = (shape[targetAxis] as number) === 1 ? 0 : index;
                offset += targetIndex * (result.strides[targetAxis] as number);
            }
        }
        result.data[offset] = (result.data[offset] as number) + (gradient.data[linear] as number);
    }
    return result;
}

/** Expande un gradiente a `shape` (el inverso de reducir un eje). */
function broadcastTo(gradient: Tensor, shape: readonly number[]): Tensor {
    const result = Tensor.zeros(shape);
    const strides = broadcastStrides(gradient.shape, gradient.strides, shape);
    for (let linear = 0; linear < result.size; linear++) {
        let remainder = linear;
        let offset = 0;
        for (let axis = 0; axis < shape.length; axis++) {
            const stride = result.strides[axis] as number;
            const index = Math.floor(remainder / stride);
            remainder -= index * stride;
            offset += index * (strides[axis] as number);
        }
        result.data[linear] = gradient.data[offset] as number;
    }
    return result;
}

/** Multiplica por un escalar sin registrar grafo. */
function scaleRaw(tensor: Tensor, factor: number): Tensor {
    const result = Tensor.zeros(tensor.shape);
    for (let index = 0; index < tensor.size; index++) {
        result.data[index] = (tensor.data[index] as number) * factor;
    }
    return result;
}

/** Operación elemento a elemento sin registrar grafo. Interno de las operaciones. */
export function elementwiseRaw(
    left: Tensor,
    right: Tensor,
    combine: (leftValue: number, rightValue: number) => number,
): Tensor {
    const shape = broadcastShapes(left.shape, right.shape);
    const leftStrides = broadcastStrides(left.shape, left.strides, shape);
    const rightStrides = broadcastStrides(right.shape, right.strides, shape);
    const result = Tensor.zeros(shape);
    for (let linear = 0; linear < result.size; linear++) {
        let remainder = linear;
        let leftOffset = 0;
        let rightOffset = 0;
        for (let axis = 0; axis < shape.length; axis++) {
            const stride = result.strides[axis] as number;
            const index = Math.floor(remainder / stride);
            remainder -= index * stride;
            leftOffset += index * (leftStrides[axis] as number);
            rightOffset += index * (rightStrides[axis] as number);
        }
        result.data[linear] = combine(
            left.data[leftOffset] as number,
            right.data[rightOffset] as number,
        );
    }
    return result;
}

/** Aplica una función elemento a elemento sin registrar grafo. Interno de las operaciones. */
export function unaryRaw(tensor: Tensor, apply: (value: number) => number): Tensor {
    const result = Tensor.zeros(tensor.shape);
    for (let index = 0; index < tensor.size; index++) {
        result.data[index] = apply(tensor.data[index] as number);
    }
    return result;
}

function unaryWithGrad(
    tensor: Tensor,
    apply: (value: number) => number,
    derivative: (value: number) => number,
): Tensor {
    const result = unaryRaw(tensor, apply);
    if (tensor.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([tensor], (gradient) => {
            const local = unaryRaw(tensor, derivative);
            tensor.addGrad(elementwiseRaw(gradient, local, (left, right) => left * right));
        });
    }
    return result;
}

/** e^x elemento a elemento. */
export function exp(tensor: Tensor): Tensor {
    return unaryWithGrad(tensor, Math.exp, Math.exp);
}

/** Logaritmo natural elemento a elemento; `log(0)` es −Infinity. */
export function log(tensor: Tensor): Tensor {
    return unaryWithGrad(tensor, Math.log, (value) => 1 / value);
}

/** ReLU: max(0, x). */
export function relu(tensor: Tensor): Tensor {
    return unaryWithGrad(tensor, (value) => Math.max(0, value), (value) => (value > 0 ? 1 : 0));
}

/** tanh elemento a elemento. */
export function tanh(tensor: Tensor): Tensor {
    return unaryWithGrad(
        tensor,
        Math.tanh,
        (value) => 1 - Math.tanh(value) * Math.tanh(value),
    );
}

/** Suma elemento a elemento, con broadcasting. */
export function add(left: Tensor, right: Tensor): Tensor {
    const result = elementwiseRaw(left, right, (leftValue, rightValue) => leftValue + rightValue);
    if (left.requiresGrad || right.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([left, right], (gradient) => {
            left.addGrad(unbroadcast(gradient, left.shape));
            right.addGrad(unbroadcast(gradient, right.shape));
        });
    }
    return result;
}

/** Resta elemento a elemento, con broadcasting. */
export function sub(left: Tensor, right: Tensor): Tensor {
    const result = elementwiseRaw(left, right, (leftValue, rightValue) => leftValue - rightValue);
    if (left.requiresGrad || right.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([left, right], (gradient) => {
            left.addGrad(unbroadcast(gradient, left.shape));
            right.addGrad(unbroadcast(scaleRaw(gradient, -1), right.shape));
        });
    }
    return result;
}

/** Producto elemento a elemento, con broadcasting. */
export function mul(left: Tensor, right: Tensor): Tensor {
    const result = elementwiseRaw(left, right, (leftValue, rightValue) => leftValue * rightValue);
    if (left.requiresGrad || right.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([left, right], (gradient) => {
            left.addGrad(
                unbroadcast(elementwiseRaw(gradient, right, (value, other) => value * other), left.shape),
            );
            right.addGrad(
                unbroadcast(elementwiseRaw(gradient, left, (value, other) => value * other), right.shape),
            );
        });
    }
    return result;
}

function matmulRaw(left: Tensor, right: Tensor): Tensor {
    if (left.rank !== 2 || right.rank !== 2) {
        throw new Error("matmul espera dos matrices de rango 2");
    }
    const rows = left.shape[0] as number;
    const inner = left.shape[1] as number;
    const columns = right.shape[1] as number;
    if (inner !== (right.shape[0] as number)) {
        throw new Error(
            `formas incompatibles para matmul: ${JSON.stringify(left.shape)} y ${JSON.stringify(right.shape)}`,
        );
    }
    const result = Tensor.zeros([rows, columns]);
    const leftRowStride = left.strides[0] as number;
    const leftColumnStride = left.strides[1] as number;
    const rightRowStride = right.strides[0] as number;
    const rightColumnStride = right.strides[1] as number;
    for (let row = 0; row < rows; row++) {
        for (let column = 0; column < columns; column++) {
            let total = 0;
            for (let step = 0; step < inner; step++) {
                total +=
                    (left.data[row * leftRowStride + step * leftColumnStride] as number) *
                    (right.data[step * rightRowStride + column * rightColumnStride] as number);
            }
            result.data[row * columns + column] = total;
        }
    }
    return result;
}

/** Producto matricial 2D: [m, k] · [k, n] → [m, n]. */
export function matmul(left: Tensor, right: Tensor): Tensor {
    const result = matmulRaw(left, right);
    if (left.requiresGrad || right.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([left, right], (gradient) => {
            left.addGrad(matmulRaw(gradient, right.transpose()));
            right.addGrad(matmulRaw(left.transpose(), gradient));
        });
    }
    return result;
}

/** Suma todos los elementos o los de un eje (ese eje desaparece del resultado). */
export function sum(tensor: Tensor, axis?: number): Tensor {
    const reducedAxis = axis;
    let result: Tensor;
    if (reducedAxis === undefined) {
        let total = 0;
        for (const value of tensor.data) total += value;
        result = Tensor.from([total], []);
    } else {
        if (!Number.isInteger(reducedAxis) || reducedAxis < 0 || reducedAxis >= tensor.rank) {
            throw new RangeError(`eje ${reducedAxis} fuera de [0, ${tensor.rank})`);
        }
        const shape = tensor.shape.filter((_, index) => index !== reducedAxis);
        result = Tensor.zeros(shape);
        for (let linear = 0; linear < tensor.size; linear++) {
            let remainder = linear;
            let outputOffset = 0;
            for (let sourceAxis = 0; sourceAxis < tensor.rank; sourceAxis++) {
                const stride = tensor.strides[sourceAxis] as number;
                const index = Math.floor(remainder / stride);
                remainder -= index * stride;
                if (sourceAxis !== reducedAxis) {
                    const outputAxis = sourceAxis < reducedAxis ? sourceAxis : sourceAxis - 1;
                    outputOffset += index * (result.strides[outputAxis] as number);
                }
            }
            result.data[outputOffset] = (result.data[outputOffset] as number) + (tensor.data[linear] as number);
        }
    }
    if (tensor.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([tensor], (gradient) => {
            const expandedShape =
                reducedAxis === undefined
                    ? tensor.shape
                    : tensor.shape.map((dimension, index) => (index === reducedAxis ? 1 : dimension));
            const expanded = reducedAxis === undefined ? gradient : gradient.reshape(expandedShape);
            tensor.addGrad(broadcastTo(expanded, tensor.shape));
        });
    }
    return result;
}

/** Media de todos los elementos o de un eje: `sum` dividido por cuántos sumó. */
export function mean(tensor: Tensor, axis?: number): Tensor {
    const total = sum(tensor, axis);
    const count = axis === undefined ? tensor.size : (tensor.shape[axis] as number);
    return mul(total, Tensor.from([1 / count], []));
}

/** Recorta columnas [start, end) de una matriz, con gradiente hacia el original. */
export function sliceColumns(tensor: Tensor, start: number, end: number): Tensor {
    if (tensor.rank !== 2) throw new Error("sliceColumns espera una matriz");
    const rows = tensor.shape[0] as number;
    const columns = tensor.shape[1] as number;
    if (!Number.isInteger(start) || !Number.isInteger(end) || start < 0 || end > columns || start >= end) {
        throw new RangeError(`columnas inválidas [${start}, ${end}) para ${columns}`);
    }
    const width = end - start;
    const data = new Float32Array(rows * width);
    for (let row = 0; row < rows; row++) {
        data.set(
            tensor.data.subarray(row * columns + start, row * columns + end),
            row * width,
        );
    }
    const result = new Tensor(data, [rows, width]);
    if (tensor.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([tensor], (gradient) => {
            const values = new Float32Array(tensor.size);
            for (let row = 0; row < rows; row++) {
                for (let column = 0; column < width; column++) {
                    values[row * columns + start + column] =
                        gradient.data[row * width + column] as number;
                }
            }
            tensor.addGrad(new Tensor(values, tensor.shape));
        });
    }
    return result;
}

/** Concatena matrices por columnas, con gradiente hacia cada una. */
export function concatColumns(tensors: readonly Tensor[]): Tensor {
    if (tensors.length === 0) throw new Error("concatColumns necesita al menos una matriz");
    const rows = tensors[0]?.shape[0] as number;
    let columns = 0;
    for (const tensor of tensors) {
        if (tensor.rank !== 2 || (tensor.shape[0] as number) !== rows) {
            throw new Error("concatColumns espera matrices con las mismas filas");
        }
        columns += tensor.shape[1] as number;
    }
    const data = new Float32Array(rows * columns);
    let offset = 0;
    for (const tensor of tensors) {
        const width = tensor.shape[1] as number;
        for (let row = 0; row < rows; row++) {
            data.set(
                tensor.data.subarray(row * width, (row + 1) * width),
                row * columns + offset,
            );
        }
        offset += width;
    }
    const result = new Tensor(data, [rows, columns]);
    if (tensors.some((tensor) => tensor.requiresGrad)) {
        result.requiresGrad = true;
        result.setGraph(tensors, (gradient) => {
            let start = 0;
            for (const tensor of tensors) {
                const width = tensor.shape[1] as number;
                const values = new Float32Array(rows * width);
                for (let row = 0; row < rows; row++) {
                    for (let column = 0; column < width; column++) {
                        values[row * width + column] =
                            gradient.data[row * columns + start + column] as number;
                    }
                }
                tensor.addGrad(new Tensor(values, [rows, width]));
                start += width;
            }
        });
    }
    return result;
}

/** Recorta filas [start, end) de una matriz, con gradiente hacia el original. */
export function sliceRows(tensor: Tensor, start: number, end: number): Tensor {
    if (tensor.rank !== 2) throw new Error("sliceRows espera una matriz");
    const rows = tensor.shape[0] as number;
    const columns = tensor.shape[1] as number;
    if (!Number.isInteger(start) || !Number.isInteger(end) || start < 0 || end > rows || start >= end) {
        throw new RangeError(`filas inválidas [${start}, ${end}) para ${rows}`);
    }
    const selected = end - start;
    const data = new Float32Array(selected * columns);
    data.set(tensor.data.subarray(start * columns, end * columns));
    const result = new Tensor(data, [selected, columns]);
    if (tensor.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([tensor], (gradient) => {
            const values = new Float32Array(tensor.size);
            values.set(gradient.data, start * columns);
            tensor.addGrad(new Tensor(values, tensor.shape));
        });
    }
    return result;
}


// Teoría: docs/teoria/nucleo/tensor.md

const isDimension = (value: number): boolean => Number.isInteger(value) && value >= 0;

const sizeOf = (shape: readonly number[]): number =>
    shape.reduce((total, dimension) => total * dimension, 1);

/** Strides canónicos (row-major): el último eje avanza de uno en uno. */
const stridesOf = (shape: readonly number[]): number[] => {
    const strides = new Array<number>(shape.length).fill(1);
    let stride = 1;
    for (let axis = shape.length - 1; axis >= 0; axis--) {
        strides[axis] = stride;
        stride *= shape[axis] as number;
    }
    return strides;
};

const checkShape = (shape: readonly number[]): void => {
    if (!shape.every(isDimension)) {
        throw new Error(`forma inválida: ${JSON.stringify(shape)}; cada dimensión debe ser un entero ≥ 0`);
    }
};

export interface TensorOptions {
    requiresGrad?: boolean;
}

/**
 * Tensor n-dimensional sobre un buffer plano (f32), con forma y strides canónicos. Sin vistas:
 * `reshape` y `transpose` devuelven tensores nuevos con sus propios datos.
 */
export class Tensor {
    readonly data: Float32Array;
    readonly shape: readonly number[];
    readonly strides: readonly number[];
    readonly size: number;
    requiresGrad: boolean;
    grad: Tensor | null = null;
    #parents: readonly Tensor[] = [];
    #backward: ((gradient: Tensor) => void) | null = null;

    constructor(data: Float32Array, shape: readonly number[], options: TensorOptions = {}) {
        checkShape(shape);
        const size = sizeOf(shape);
        if (data.length !== size) {
            throw new Error(
                `los datos tienen ${data.length} elementos y la forma ${JSON.stringify(shape)} pide ${size}`,
            );
        }
        this.data = data;
        this.shape = [...shape];
        this.strides = stridesOf(shape);
        this.size = size;
        this.requiresGrad = options.requiresGrad ?? false;
    }

    get rank(): number {
        return this.shape.length;
    }

    static zeros(shape: readonly number[], options: TensorOptions = {}): Tensor {
        checkShape(shape);
        return new Tensor(new Float32Array(sizeOf(shape)), shape, options);
    }

    static ones(shape: readonly number[], options: TensorOptions = {}): Tensor {
        const tensor = Tensor.zeros(shape, options);
        tensor.data.fill(1);
        return tensor;
    }

    static from(values: ArrayLike<number>, shape?: readonly number[], options: TensorOptions = {}): Tensor {
        const data = Float32Array.from(values);
        return new Tensor(data, shape ?? [data.length], options);
    }

    get(...indices: number[]): number {
        return this.data[this.offsetOf(indices)] as number;
    }

    set(value: number, ...indices: number[]): void {
        this.data[this.offsetOf(indices)] = value;
    }

    reshape(shape: readonly number[]): Tensor {
        checkShape(shape);
        const size = sizeOf(shape);
        if (size !== this.size) {
            throw new Error(
                `no se puede pasar de ${JSON.stringify(this.shape)} (${this.size}) a ${JSON.stringify(shape)} (${size})`,
            );
        }
        return new Tensor(this.data.slice(), shape);
    }

    transpose(axes?: readonly number[]): Tensor {
        const order = axes ?? [...this.shape.keys()].reverse();
        if (order.length !== this.rank) {
            throw new Error(`el orden debe tener ${this.rank} ejes, no ${order.length}`);
        }
        const seen = new Set<number>();
        for (const axis of order) {
            if (!Number.isInteger(axis) || axis < 0 || axis >= this.rank || seen.has(axis)) {
                throw new Error(`orden inválido: ${JSON.stringify(order)}`);
            }
            seen.add(axis);
        }
        const shape = order.map((axis) => this.shape[axis] as number);
        const result = new Tensor(new Float32Array(this.size), shape);
        for (let linear = 0; linear < this.size; linear++) {
            let remainder = linear;
            let sourceOffset = 0;
            for (let axis = 0; axis < order.length; axis++) {
                const stride = result.strides[axis] as number;
                const index = Math.floor(remainder / stride);
                remainder -= index * stride;
                sourceOffset += index * (this.strides[order[axis] as number] as number);
            }
            result.data[linear] = this.data[sourceOffset] as number;
        }
        return result;
    }

    /** Engancha el resultado al grafo. Interno de las operaciones. */
    setGraph(parents: readonly Tensor[], backward: (gradient: Tensor) => void): void {
        this.#parents = parents;
        this.#backward = backward;
    }

    /** Acumula un gradiente con la forma de este tensor. Interno de las operaciones. */
    addGrad(gradient: Tensor): void {
        if (!this.requiresGrad) return;
        const sameShape =
            gradient.shape.length === this.shape.length &&
            gradient.shape.every((dimension, axis) => dimension === this.shape[axis]);
        if (!sameShape) {
            throw new Error("el gradiente no tiene la forma del tensor");
        }
        if (this.grad === null) {
            this.grad = new Tensor(gradient.data.slice(), gradient.shape);
            return;
        }
        for (let index = 0; index < this.grad.data.length; index++) {
            this.grad.data[index] = (this.grad.data[index] as number) + (gradient.data[index] as number);
        }
    }

    zeroGrad(): void {
        this.grad = null;
    }

    /** Retropropaga desde este tensor (semilla de unos) por todo el grafo. */
    backward(): void {
        const order: Tensor[] = [];
        const visited = new Set<Tensor>();
        const visit = (tensor: Tensor): void => {
            if (visited.has(tensor)) return;
            visited.add(tensor);
            for (const parent of tensor.#parents) visit(parent);
            order.push(tensor);
        };
        visit(this);
        this.grad = Tensor.ones(this.shape);
        for (let index = order.length - 1; index >= 0; index--) {
            const tensor = order[index] as Tensor;
            const gradient = tensor.grad;
            if (gradient && tensor.#backward) tensor.#backward(gradient);
        }
    }

    private offsetOf(indices: readonly number[]): number {
        if (indices.length !== this.rank) {
            throw new Error(`se esperaban ${this.rank} índices, no ${indices.length}`);
        }
        let offset = 0;
        for (let axis = 0; axis < indices.length; axis++) {
            const index = indices[axis] as number;
            const dimension = this.shape[axis] as number;
            if (!Number.isInteger(index) || index < 0 || index >= dimension) {
                throw new RangeError(`índice ${index} fuera de [0, ${dimension}) en el eje ${axis}`);
            }
            offset += index * (this.strides[axis] as number);
        }
        return offset;
    }
}

import { readFileSync, renameSync, writeFileSync } from "node:fs";

import { BPETokenizer } from "./bpe.ts";
import type { Merge } from "./bpe.ts";

export const BPE_FORMAT = "ailib-ts.bpe";
export const BPE_VERSION = 1;

/** Guarda los merges de un tokenizer BPE en JSON versionado. */
export function saveBPE(path: string, tokenizer: BPETokenizer): void {
    const payload = {
        format: BPE_FORMAT,
        version: BPE_VERSION,
        merges: tokenizer.merges,
    };
    const temporary = `${path}.tmp`;
    writeFileSync(temporary, `${JSON.stringify(payload)}\n`, "utf8");
    renameSync(temporary, path);
}

/** Carga un tokenizer BPE desde un JSON guardado con `saveBPE`. */
export function loadBPE(path: string): BPETokenizer {
    let parsed: unknown;
    try {
        parsed = JSON.parse(readFileSync(path, "utf8"));
    } catch {
        throw new Error(`no se puede leer el tokenizer BPE ${path}`);
    }
    if (typeof parsed !== "object" || parsed === null) {
        throw new Error(`${path}: BPE inválido`);
    }
    const record = parsed as Record<string, unknown>;
    if (record.format !== BPE_FORMAT || record.version !== BPE_VERSION) {
        throw new Error(`${path}: formato o versión de BPE inválidos`);
    }
    if (!Array.isArray(record.merges)) throw new Error(`${path}: faltan los merges`);
    const tokenizer = new BPETokenizer();
    tokenizer.restore(record.merges as Merge[]);
    return tokenizer;
}

import { ByteTokenizer } from "./tokenizer.ts";

export interface BPEOptions {
    vocabularySize?: number;
    minFrequency?: number;
}

export interface Merge {
    left: number;
    right: number;
    token: number;
}

/** Byte Pair Encoding: parte de 256 bytes y fusiona pares frecuentes. */
export class BPETokenizer {
    readonly base = new ByteTokenizer();
    readonly #merges: Merge[] = [];
    #tokenBytes: Uint8Array[];
    #vocabularySize = 256;

    constructor() {
        this.#tokenBytes = Array.from({ length: 256 }, (_, byte) => new Uint8Array([byte]));
    }

    get vocabularySize(): number {
        return this.#vocabularySize;
    }

    get merges(): readonly Merge[] {
        return this.#merges;
    }

    /** Aprende merges sobre `text` hasta el vocabulario pedido. */
    train(text: string, options: BPEOptions = {}): void {
        const target = options.vocabularySize ?? 512;
        const minFrequency = options.minFrequency ?? 2;
        if (!Number.isInteger(target) || target < 256) {
            throw new RangeError(`el vocabulario debe ser al menos 256, no ${target}`);
        }
        if (!Number.isInteger(minFrequency) || minFrequency < 1) {
            throw new RangeError(`la frecuencia mínima debe ser positiva, no ${minFrequency}`);
        }
        this.#merges.length = 0;
        this.#tokenBytes = Array.from({ length: 256 }, (_, byte) => new Uint8Array([byte]));
        this.#vocabularySize = 256;

        let sequence = Array.from(this.base.encode(text));
        let training = true;
        while (training && this.#vocabularySize < target && sequence.length >= 2) {
            const counts = new Map<string, number>();
            for (let index = 0; index + 1 < sequence.length; index++) {
                const key = `${sequence[index]},${sequence[index + 1]}`;
                counts.set(key, (counts.get(key) ?? 0) + 1);
            }
            let bestKey: string | null = null;
            let bestCount = 0;
            for (const [key, count] of counts) {
                if (
                    count > bestCount ||
                    (count === bestCount && (bestKey === null || key < bestKey))
                ) {
                    bestKey = key;
                    bestCount = count;
                }
            }
            if (bestKey === null || bestCount < minFrequency) {
                training = false;
            } else {
                const [left, right] = bestKey.split(",").map(Number) as [number, number];
                const token = this.#vocabularySize;
                const first = this.#tokenBytes[left] as Uint8Array;
                const second = this.#tokenBytes[right] as Uint8Array;
                const bytes = new Uint8Array(first.length + second.length);
                bytes.set(first);
                bytes.set(second, first.length);
                this.#merges.push({ left, right, token });
                this.#tokenBytes.push(bytes);
                this.#vocabularySize += 1;

                const merged: number[] = [];
                for (let index = 0; index < sequence.length; ) {
                    if (
                        index + 1 < sequence.length &&
                        sequence[index] === left &&
                        sequence[index + 1] === right
                    ) {
                        merged.push(token);
                        index += 2;
                    } else {
                        merged.push(sequence[index] as number);
                        index += 1;
                    }
                }
                sequence = merged;
            }
        }
    }

    encode(text: string): number[] {
        let sequence = Array.from(this.base.encode(text));
        for (const merge of this.#merges) {
            const merged: number[] = [];
            for (let index = 0; index < sequence.length; ) {
                if (
                    index + 1 < sequence.length &&
                    sequence[index] === merge.left &&
                    sequence[index + 1] === merge.right
                ) {
                    merged.push(merge.token);
                    index += 2;
                } else {
                    merged.push(sequence[index] as number);
                    index += 1;
                }
            }
            sequence = merged;
        }
        return sequence;
    }

    /** Reconstruye el tokenizer a partir de merges ya aprendidos. */
    restore(merges: readonly Merge[]): void {
        this.#merges.length = 0;
        this.#tokenBytes = Array.from({ length: 256 }, (_, byte) => new Uint8Array([byte]));
        this.#vocabularySize = 256;
        for (const merge of merges) {
            if (
                !Number.isInteger(merge.left) ||
                !Number.isInteger(merge.right) ||
                merge.token !== this.#vocabularySize ||
                merge.left < 0 ||
                merge.right < 0 ||
                merge.left >= this.#vocabularySize ||
                merge.right >= this.#vocabularySize
            ) {
                throw new Error(`merge inválido para el token ${this.#vocabularySize}`);
            }
            const first = this.#tokenBytes[merge.left] as Uint8Array;
            const second = this.#tokenBytes[merge.right] as Uint8Array;
            const bytes = new Uint8Array(first.length + second.length);
            bytes.set(first);
            bytes.set(second, first.length);
            this.#merges.push({ left: merge.left, right: merge.right, token: merge.token });
            this.#tokenBytes.push(bytes);
            this.#vocabularySize += 1;
        }
    }

    decode(tokens: readonly number[]): string {
        const bytes: number[] = [];
        for (const token of tokens) {
            if (!Number.isInteger(token) || token < 0 || token >= this.#vocabularySize) {
                throw new RangeError(`token inválido ${token}; fuera del vocabulario`);
            }
            const chunk = this.#tokenBytes[token] as Uint8Array;
            for (const byte of chunk) bytes.push(byte);
        }
        return this.base.decode(bytes);
    }
}

import { ByteTokenizer } from "./tokenizer.ts";

export interface EntropyPatchOptions {
    threshold?: number;
    maxPatch?: number;
    vocabularySize?: number;
}

const keyOf = (bytes: Uint8Array): string => [...bytes].join(",");

/**
 * Patcher BLT-style: agrupa bytes mientras la entropía del siguiente byte es baja.
 * No es BLT completo: no aprende un modelo de parches, usa un n-gram de bytes.
 */
export class EntropyByteTokenizer {
    readonly #bytes = new ByteTokenizer();
    readonly #counts = new Map<string, Map<number, number>>();
    readonly #patches: Uint8Array[] = [];
    readonly #patchIds = new Map<string, number>();
    #threshold = 2;
    #maxPatch = 8;

    get vocabularySize(): number {
        return 256 + this.#patches.length;
    }

    train(text: string, options: EntropyPatchOptions = {}): void {
        this.#threshold = options.threshold ?? 2;
        this.#maxPatch = options.maxPatch ?? 8;
        const vocabularySize = options.vocabularySize ?? 2048;
        this.#counts.clear();
        this.#patches.length = 0;
        this.#patchIds.clear();
        const bytes = Uint8Array.from(this.#bytes.encode(text));
        for (let position = 0; position < bytes.length; position++) {
            const context = this.#context(bytes, position);
            let distribution = this.#counts.get(context);
            if (!distribution) {
                distribution = new Map();
                this.#counts.set(context, distribution);
            }
            distribution.set(bytes[position] as number, (distribution.get(bytes[position] as number) ?? 0) + 1);
        }
        const patchCounts = new Map<string, { bytes: Uint8Array; count: number }>();
        for (const patch of this.#patch(bytes)) {
            const key = keyOf(patch);
            const current = patchCounts.get(key);
            patchCounts.set(key, { bytes: patch, count: (current?.count ?? 0) + 1 });
        }
        const chosen = [...patchCounts.entries()]
            .filter(([key]) => key.includes(","))
            .sort((first, second) => second[1].count - first[1].count || first[0].localeCompare(second[0]))
            .slice(0, Math.max(0, vocabularySize - 256));
        for (const [key, value] of chosen) {
            this.#patchIds.set(key, 256 + this.#patches.length);
            this.#patches.push(value.bytes);
        }
    }

    #context(bytes: Uint8Array, position: number): string {
        if (position === 0) return "^,^";
        if (position === 1) return `^,${bytes[0]}`;
        return `${bytes[position - 2]},${bytes[position - 1]}`;
    }

    #entropy(context: string): number {
        const distribution = this.#counts.get(context);
        if (!distribution) return 8;
        let total = 0;
        for (const count of distribution.values()) total += count;
        let entropy = 0;
        for (const count of distribution.values()) {
            const probability = count / total;
            entropy -= probability * Math.log2(probability);
        }
        return entropy;
    }

    #patch(bytes: Uint8Array): Uint8Array[] {
        const patches: Uint8Array[] = [];
        let start = 0;
        while (start < bytes.length) {
            let end = start + 1;
            while (
                end < bytes.length &&
                end - start < this.#maxPatch &&
                this.#entropy(this.#context(bytes, end)) < this.#threshold
            ) {
                end += 1;
            }
            patches.push(Uint8Array.from(bytes.slice(start, end)));
            start = end;
        }
        return patches;
    }

    encode(text: string): number[] {
        const output: number[] = [];
        for (const patch of this.#patch(Uint8Array.from(this.#bytes.encode(text)))) {
            const key = keyOf(patch);
            const id = this.#patchIds.get(key);
            if (id !== undefined) {
                output.push(id);
            } else {
                output.push(...patch);
            }
        }
        return output;
    }

    decode(tokens: readonly number[]): string {
        const bytes: number[] = [];
        for (const token of tokens) {
            if (token < 256) {
                bytes.push(token);
            } else {
                const patch = this.#patches[token - 256];
                if (!patch) throw new RangeError(`token fuera de vocabulario: ${token}`);
                bytes.push(...patch);
            }
        }
        return this.#bytes.decode(bytes);
    }
}

/** Tokenizer por caracteres Unicode (code points), con `<unk>` para lo no visto. */
export class CharTokenizer {
    #characters: string[] = ["<unk>"];
    #index = new Map<string, number>([["<unk>", 0]]);

    get vocabularySize(): number {
        return this.#characters.length;
    }

    train(text: string): this {
        if (text.length === 0) throw new Error("no se puede entrenar con texto vacío");
        const counts = new Map<string, number>();
        for (const character of text) {
            counts.set(character, (counts.get(character) ?? 0) + 1);
        }
        const ordered = [...counts.entries()]
            .sort((first, second) => second[1] - first[1] || first[0].localeCompare(second[0]))
            .map(([character]) => character);
        this.#characters = ["<unk>", ...ordered];
        this.#index = new Map(this.#characters.map((character, index) => [character, index]));
        return this;
    }

    encode(text: string): number[] {
        const tokens: number[] = [];
        for (const character of text) {
            tokens.push(this.#index.get(character) ?? 0);
        }
        return tokens;
    }

    decode(tokens: readonly number[]): string {
        let text = "";
        for (const token of tokens) {
            if (!Number.isInteger(token) || token < 0 || token >= this.#characters.length) {
                throw new RangeError(`token fuera de vocabulario: ${token}`);
            }
            text += token === 0 ? "\uFFFD" : (this.#characters[token] as string);
        }
        return text;
    }
}

export interface ContextTokenizerOptions {
    leftContext?: number;
    rightContext?: number;
    lookahead?: "chars" | "word";
    penalty?: number;
}

type ByteClass = "letter" | "digit" | "space" | "other";

function classOf(byte: number): ByteClass {
    if (byte === 32 || byte === 9 || byte === 10 || byte === 13) return "space";
    if (byte >= 48 && byte <= 57) return "digit";
    if ((byte >= 65 && byte <= 90) || (byte >= 97 && byte <= 122) || byte >= 128) return "letter";
    return "other";
}

/**
 * Tokenizer lossless con segmentación por programación dinámica. El coste de un corte depende del
 * contexto izquierdo/derecho: partir una palabra se penaliza; `lookahead: "word"` mira hasta el final
 * de la palabra siguiente.
 */
export class ContextTokenizer {
    readonly #tokens: Uint8Array[];
    readonly #root: TrieNode = { children: new Map() };
    readonly #left: number;
    readonly #right: number;
    readonly #lookahead: "chars" | "word";
    readonly #penalty: number;

    constructor(tokens: readonly Uint8Array[], options: ContextTokenizerOptions = {}) {
        if (tokens.length === 0) throw new Error("hacen falta tokens");
        this.#tokens = [...tokens];
        this.#left = options.leftContext ?? 2;
        this.#right = options.rightContext ?? 2;
        this.#lookahead = options.lookahead ?? "chars";
        this.#penalty = options.penalty ?? 0.5;
        if (!Number.isInteger(this.#left) || this.#left < 0 || this.#left > 3) {
            throw new Error(`leftContext inválido: ${this.#left}`);
        }
        if (!Number.isInteger(this.#right) || this.#right < 0 || this.#right > 3) {
            throw new Error(`rightContext inválido: ${this.#right}`);
        }
        if (!Number.isFinite(this.#penalty) || this.#penalty < 0) {
            throw new Error(`penalty inválida: ${this.#penalty}`);
        }
        for (let id = 0; id < this.#tokens.length; id++) {
            let node = this.#root;
            for (const byte of this.#tokens[id] as Uint8Array) {
                let child = node.children.get(byte);
                if (!child) {
                    child = { children: new Map() };
                    node.children.set(byte, child);
                }
                node = child;
            }
            node.token = id;
        }
    }

    get vocabularySize(): number {
        return this.#tokens.length;
    }

    encode(text: string): number[] {
        const bytes = new TextEncoder().encode(text);
        const size = bytes.length;
        const costs = new Float64Array(size + 1).fill(Number.POSITIVE_INFINITY);
        const previous = new Int32Array(size + 1).fill(-1);
        const chosen = new Int32Array(size + 1).fill(-1);
        costs[0] = 0;
        for (let position = 0; position < size; position++) {
            const base = costs[position] as number;
            if (Number.isFinite(base)) {
                let node = this.#root;
                let end = position;
                let searching = true;
                while (end < size && searching) {
                    const child = node.children.get(bytes[end] as number);
                    if (!child) {
                        searching = false;
                    } else {
                        node = child;
                        if (node.token !== undefined) {
                            const cost = base + 1 + this.#boundaryPenalty(bytes, position, end + 1);
                            if (cost < (costs[end + 1] as number)) {
                                costs[end + 1] = cost;
                                previous[end + 1] = position;
                                chosen[end + 1] = node.token;
                            }
                        }
                        end += 1;
                    }
                }
            }
        }
        if (!Number.isFinite(costs[size] as number)) {
            throw new Error("no hay segmentación con los tokens dados");
        }
        const tokens: number[] = [];
        let cursor = size;
        while (cursor > 0) {
            tokens.push(chosen[cursor] as number);
            cursor = previous[cursor] as number;
        }
        return tokens.reverse();
    }

    decode(tokens: readonly number[]): string {
        const bytes: number[] = [];
        for (const token of tokens) {
            if (!Number.isInteger(token) || token < 0 || token >= this.#tokens.length) {
                throw new RangeError(`token fuera de vocabulario: ${token}`);
            }
            for (const byte of this.#tokens[token] as Uint8Array) bytes.push(byte);
        }
        return new TextDecoder().decode(Uint8Array.from(bytes));
    }

    #boundaryPenalty(bytes: Uint8Array, start: number, end: number): number {
        let penalty = 0;
        const first = classOf(bytes[start] as number);
        const last = classOf(bytes[end - 1] as number);
        if (this.#left > 0 && start > 0 && first === "letter") {
            const before = classOf(bytes[start - 1] as number);
            if (before === "letter") penalty += this.#penalty * (this.#left / 3);
        }
        if (this.#right > 0 && end < bytes.length && last === "letter") {
            const after = classOf(bytes[end] as number);
            if (after === "letter") penalty += this.#penalty * (this.#right / 3);
        }
        if (this.#lookahead === "word" && end < bytes.length && last === "letter") {
            let wordEnd = end;
            while (wordEnd < bytes.length && classOf(bytes[wordEnd] as number) === "letter") {
                wordEnd += 1;
            }
            if (wordEnd > end) penalty += this.#penalty * ((wordEnd - end) / Math.max(1, wordEnd - start));
        }
        return penalty;
    }
}

/**
 * Tokenizer cuyos tokens son secuencias de bytes. Empieza con los 256 bytes, así que siempre puede
 * reconstruir cualquier texto; los tokens adicionales solo acortan la secuencia.
 */
export class LosslessByteTokenizer {
    readonly #tokens: Uint8Array[] = [];
    readonly #ids = new Map<string, number>();
    readonly #root: TrieNode = { children: new Map() };
    readonly #maxTokens: number;

    constructor(maxTokens = 4096) {
        if (!Number.isInteger(maxTokens) || maxTokens < 256) {
            throw new Error(`maxTokens debe ser un entero ≥ 256, no ${maxTokens}`);
        }
        this.#maxTokens = maxTokens;
        for (let byte = 0; byte < 256; byte++) {
            this.add(new Uint8Array([byte]));
        }
    }

    get vocabularySize(): number {
        return this.#tokens.length;
    }

    add(bytes: Uint8Array): number | null {
        if (bytes.length === 0) throw new Error("un token no puede estar vacío");
        const key = String.fromCharCode(...bytes);
        const existing = this.#ids.get(key);
        if (existing !== undefined) return existing;
        if (this.#tokens.length >= this.#maxTokens) return null;
        const id = this.#tokens.length;
        this.#tokens.push(bytes);
        this.#ids.set(key, id);
        let node = this.#root;
        for (const byte of bytes) {
            let child = node.children.get(byte);
            if (!child) {
                child = { children: new Map() };
                node.children.set(byte, child);
            }
            node = child;
        }
        node.token = id;
        return id;
    }

    encode(text: string): number[] {
        const bytes = new TextEncoder().encode(text);
        const tokens: number[] = [];
        let position = 0;
        while (position < bytes.length) {
            let node = this.#root;
            let matched = -1;
            let matchedEnd = position;
            let index = position;
            let searching = true;
            while (index < bytes.length && searching) {
                const child = node.children.get(bytes[index] as number);
                if (!child) {
                    searching = false;
                } else {
                    node = child;
                    if (node.token !== undefined) {
                        matched = node.token;
                        matchedEnd = index + 1;
                    }
                    index += 1;
                }
            }
            if (matched >= 0) {
                tokens.push(matched);
                position = matchedEnd;
            } else {
                tokens.push(bytes[position] as number);
                position += 1;
            }
        }
        return tokens;
    }

    decode(tokens: readonly number[]): string {
        const bytes = new Uint8Array(
            tokens.reduce((total, token) => {
                if (!Number.isInteger(token) || token < 0 || token >= this.#tokens.length) {
                    throw new RangeError(`token fuera de vocabulario: ${token}`);
                }
                return total + (this.#tokens[token] as Uint8Array).length;
            }, 0),
        );
        let offset = 0;
        for (const token of tokens) {
            const bytesToken = this.#tokens[token] as Uint8Array;
            bytes.set(bytesToken, offset);
            offset += bytesToken.length;
        }
        return new TextDecoder().decode(bytes);
    }
}

import { BPETokenizer } from "./bpe.ts";

const MARKER = "\u2581";
const ESCAPE = "\uE000";
const ESCAPED_MARKER = "M";

export interface SentencePieceOptions {
    vocabularySize?: number;
    minFrequency?: number;
}

/**
 * SentencePiece-BPE lossless: flujo crudo con ▁ para espacios.
 * El marcador literal se escapa como ▁▁ para que decode sea reversible.
 */
export class SentencePieceTokenizer {
    readonly #bpe = new BPETokenizer();

    get vocabularySize(): number {
        return this.#bpe.vocabularySize;
    }

    train(text: string, options: SentencePieceOptions = {}): void {
        this.#bpe.train(this.#normalize(text), {
            vocabularySize: options.vocabularySize ?? 2048,
            minFrequency: options.minFrequency ?? 2,
        });
    }

    #normalize(text: string): string {
        return text
            .replaceAll(ESCAPE, `${ESCAPE}${ESCAPE}`)
            .replaceAll(MARKER, `${ESCAPE}${ESCAPED_MARKER}`)
            .replaceAll(" ", MARKER);
    }

    #restore(text: string): string {
        let output = "";
        for (let index = 0; index < text.length; index++) {
            const character = text[index] as string;
            if (character === ESCAPE) {
                const next = text[index + 1];
                if (next === ESCAPE) {
                    output += ESCAPE;
                    index += 1;
                } else if (next === ESCAPED_MARKER) {
                    output += MARKER;
                    index += 1;
                } else {
                    output += ESCAPE;
                }
            } else if (character === MARKER) {
                output += " ";
            } else {
                output += character;
            }
        }
        return output;
    }

    encode(text: string): number[] {
        return this.#bpe.encode(this.#normalize(text));
    }

    decode(tokens: readonly number[]): string {
        return this.#restore(this.#bpe.decode(tokens));
    }
}

import { BPETokenizer } from "./bpe.ts";
import { ByteTokenizer } from "./tokenizer.ts";
import { LosslessByteTokenizer } from "./lossless.ts";

export type TokenizerFamily = "byte" | "char" | "word" | "dict" | "bpe";

export interface TokenizerProgram {
    family: TokenizerFamily;
    vocabularySize: number;
    maxTokenBytes: number;
    minFrequency: number;
}

export interface ProgramTokenizer {
    readonly vocabularySize: number;
    encode(text: string): number[];
    decode(tokens: readonly number[]): string;
}

const FAMILIES: TokenizerFamily[] = ["byte", "char", "word", "dict", "bpe"];
const VOCABULARIES = [512, 1024, 2048, 4096, 8192, 16384, 32768];
const LENGTHS = [32, 64, 128];
const FREQUENCIES = [1, 2, 5, 10];

/** Genoma mínimo de F9.1; el mapeo GE/SGE/πGE llega en F9.2. */
export function programFromGenome(genome: readonly number[]): TokenizerProgram {
    if (genome.length < 4) throw new Error("el genoma necesita al menos 4 codones");
    return {
        family: FAMILIES[(genome[0] as number) % FAMILIES.length] as TokenizerFamily,
        vocabularySize: VOCABULARIES[(genome[1] as number) % VOCABULARIES.length] as number,
        maxTokenBytes: LENGTHS[(genome[2] as number) % LENGTHS.length] as number,
        minFrequency: FREQUENCIES[(genome[3] as number) % FREQUENCIES.length] as number,
    };
}

function units(text: string): string[] {
    return text.match(/\p{L}+|\p{N}+|[^\p{L}\p{N}\s]+|\s+/gu) ?? [];
}

function byteLength(value: string): number {
    return new TextEncoder().encode(value).length;
}

function addByFrequency(
    tokenizer: LosslessByteTokenizer,
    counts: Map<string, number>,
    minFrequency: number,
    maxTokenBytes: number,
): void {
    const ordered = [...counts.entries()]
        .filter(([value, count]) => count >= minFrequency && byteLength(value) <= maxTokenBytes)
        .sort((first, second) => second[1] - first[1] || first[0].localeCompare(second[0]));
    for (const [value] of ordered) {
        if (tokenizer.add(new TextEncoder().encode(value)) === null) return;
    }
}

/** Construye el tokenizer que describe el programa. */
export function buildTokenizer(program: TokenizerProgram, train: string): ProgramTokenizer {
    if (program.family === "byte") return new ByteTokenizer();
    if (program.family === "bpe") {
        const bpe = new BPETokenizer();
        bpe.train(train, {
            vocabularySize: Math.max(256, program.vocabularySize),
            minFrequency: program.minFrequency,
        });
        return bpe;
    }

    const tokenizer = new LosslessByteTokenizer(program.vocabularySize);
    if (program.family === "char") {
        const counts = new Map<string, number>();
        for (const character of train) {
            counts.set(character, (counts.get(character) ?? 0) + 1);
        }
        addByFrequency(tokenizer, counts, 1, program.maxTokenBytes);
        return tokenizer;
    }

    if (program.family === "word") {
        const counts = new Map<string, number>();
        for (const unit of units(train)) {
            counts.set(unit, (counts.get(unit) ?? 0) + 1);
        }
        addByFrequency(tokenizer, counts, program.minFrequency, program.maxTokenBytes);
        return tokenizer;
    }

    const counts = new Map<string, number>();
    for (const unit of units(train)) {
        const characters = [...unit];
        for (let start = 0; start < characters.length; start++) {
            let candidate = "";
            for (
                let end = start;
                end < characters.length &&
                byteLength(candidate + characters[end]) <= program.maxTokenBytes;
                end++
            ) {
                candidate += characters[end] as string;
                counts.set(candidate, (counts.get(candidate) ?? 0) + 1);
            }
        }
    }
    addByFrequency(tokenizer, counts, program.minFrequency, program.maxTokenBytes);
    return tokenizer;
}

export function isLossless(tokenizer: ProgramTokenizer, texts: readonly string[]): boolean {
    return texts.every((text) => tokenizer.decode(tokenizer.encode(text)) === text);
}

export interface Tokenizer {
    readonly vocabularySize: number;
    encode(text: string): number[];
    decode(tokens: readonly number[]): string;
}

const encoder = new TextEncoder();
const decoder = new TextDecoder();

/** Tokenizer byte: cada token es un byte UTF-8 (0..255). */
export class ByteTokenizer implements Tokenizer {
    readonly vocabularySize = 256;

    encode(text: string): number[] {
        return Array.from(encoder.encode(text));
    }

    decode(tokens: readonly number[]): string {
        const bytes = new Uint8Array(tokens.length);
        tokens.forEach((token, index) => {
            if (!Number.isInteger(token) || token < 0 || token >= 256) {
                throw new RangeError(`token inválido ${token} en ${index}; debe ser un byte 0..255`);
            }
            bytes[index] = token;
        });
        return decoder.decode(bytes);
    }
}

const UNIT = /\p{L}+|\p{N}+|[^\p{L}\p{N}\s]+|\s+/gu;
const unitLike = (value: string): boolean => /^[\p{L}\p{N}]+$/u.test(value);

export interface UnigramOptions {
    vocabularySize?: number;
    maxTokenLength?: number;
    iterations?: number;
    minFrequency?: number;
}

/**
 * Unigram lossless: candidatos por substrings, EM de probabilidades y Viterbi.
 * Los bytes no vistos caen a un fallback de 256 bytes.
 */
export class UnigramTokenizer {
    #tokens: string[] = [];
    #probs: number[] = [];
    #index = new Map<string, number>();
    #byteStart = 0;
    #maxTokenLength = 16;

    get vocabularySize(): number {
        return this.#tokens.length + (this.#byteStart === 0 ? 0 : 256);
    }

    train(text: string, options: UnigramOptions = {}): void {
        const target = options.vocabularySize ?? 2048;
        const maxTokenLength = options.maxTokenLength ?? 16;
        const iterations = options.iterations ?? 2;
        const minFrequency = options.minFrequency ?? 2;
        if (!Number.isInteger(target) || target < 256) {
            throw new RangeError(`el vocabulario debe ser al menos 256, no ${target}`);
        }
        if (!Number.isInteger(maxTokenLength) || maxTokenLength < 1) {
            throw new RangeError(`maxTokenLength inválido: ${maxTokenLength}`);
        }
        this.#maxTokenLength = maxTokenLength;
        this.#tokens = [];
        this.#probs = [];
        this.#index.clear();
        this.#byteStart = 0;

        const units = text.match(UNIT) ?? [];
        const counts = new Map<string, number>();
        for (const unit of units) {
            if (!unitLike(unit)) {
                counts.set(unit, (counts.get(unit) ?? 0) + 1);
            } else {
                const characters = [...unit];
                for (let start = 0; start < characters.length; start++) {
                    let candidate = "";
                    for (
                        let end = start;
                        end < characters.length && end - start < maxTokenLength;
                        end++
                    ) {
                        candidate += characters[end] as string;
                        counts.set(candidate, (counts.get(candidate) ?? 0) + 1);
                    }
                }
            }
        }
        const chars = [...new Set([...counts.keys()].filter((value) => [...value].length === 1))];
        const candidates = [...counts.entries()]
            .filter(([value, count]) => count >= minFrequency && [...value].length > 1)
            .sort((first, second) => second[1] - first[1] || first[0].localeCompare(second[0]))
            .slice(0, Math.max(0, target - chars.length))
            .map(([value]) => value);
        for (const value of [...chars, ...candidates]) this.#add(value, counts.get(value) ?? 1);

        for (let iteration = 0; iteration < iterations; iteration++) {
            const selected = new Array<number>(this.#tokens.length).fill(0.5);
            for (const unit of units) {
                if (unitLike(unit)) {
                    for (const token of this.#viterbi(unit)) selected[token] = (selected[token] as number) + 1;
                } else {
                    const id = this.#index.get(unit);
                    if (id !== undefined) selected[id] = (selected[id] as number) + 1;
                }
            }
            const total = selected.reduce((sum, value) => sum + value, 0);
            this.#probs = selected.map((value) => value / total);
        }

        this.#byteStart = this.#tokens.length;
    }

    #add(value: string, count: number): void {
        const id = this.#tokens.length;
        this.#tokens.push(value);
        this.#probs.push(count);
        this.#index.set(value, id);
    }

    #viterbi(unit: string): number[] {
        const characters = [...unit];
        const size = characters.length;
        const costs = new Array<number>(size + 1).fill(Number.POSITIVE_INFINITY);
        const previous = new Array<number>(size + 1).fill(-1);
        const chosen = new Array<number>(size + 1).fill(-1);
        costs[0] = 0;
        for (let start = 0; start < size; start++) {
            const base = costs[start] as number;
            if (Number.isFinite(base)) {
                let candidate = "";
                for (
                    let end = start;
                    end < size && end - start < this.#maxTokenLength;
                    end++
                ) {
                    candidate += characters[end] as string;
                    const id = this.#index.get(candidate);
                    if (id !== undefined) {
                        const price = base - Math.log(Math.max(this.#probs[id] as number, 1e-9));
                        if (price < (costs[end + 1] as number)) {
                            costs[end + 1] = price;
                            previous[end + 1] = start;
                            chosen[end + 1] = id;
                        }
                    }
                }
            }
        }
        if (!Number.isFinite(costs[size] as number)) return [];
        const tokens: number[] = [];
        let cursor = size;
        while (cursor > 0) {
            tokens.push(chosen[cursor] as number);
            cursor = previous[cursor] as number;
        }
        return tokens.reverse();
    }

    encode(text: string): number[] {
        const output: number[] = [];
        for (const unit of text.match(UNIT) ?? []) {
            if (unitLike(unit)) {
                const tokens = this.#viterbi(unit);
                if (tokens.length > 0) {
                    output.push(...tokens);
                } else {
                    output.push(...this.#fallback(unit));
                }
            } else {
                const id = this.#index.get(unit);
                if (id !== undefined) output.push(id);
                else output.push(...this.#fallback(unit));
            }
        }
        return output;
    }

    #fallback(text: string): number[] {
        return [...new TextEncoder().encode(text)].map((byte) => this.#byteStart + byte);
    }

    decode(tokens: readonly number[]): string {
        const bytes: number[] = [];
        for (const token of tokens) {
            if (token < this.#byteStart) {
                bytes.push(...new TextEncoder().encode(this.#tokens[token] as string));
            } else if (token < this.#byteStart + 256) {
                bytes.push(token - this.#byteStart);
            } else {
                throw new RangeError(`token fuera de vocabulario: ${token}`);
            }
        }
        return new TextDecoder().decode(Uint8Array.from(bytes));
    }
}

export interface WordTokenizerOptions {

}

/** Tokenizer por palabras, números, signos y espacios; lo no visto cae en `<unk>`. */
export class WordTokenizer {
    #tokens: string[] = ["<unk>"];
    #index = new Map<string, number>([["<unk>", 0]]);

    get vocabularySize(): number {
        return this.#tokens.length;
    }

    train(text: string, options: WordTokenizerOptions = {}): this {
        if (text.length === 0) throw new Error("no se puede entrenar con texto vacío");
        const minFrequency = options.minFrequency ?? 2;
        if (!Number.isInteger(minFrequency) || minFrequency <= 0) {
            throw new Error(`frecuencia mínima inválida: ${minFrequency}`);
        }
        const counts = new Map<string, number>();
        for (const unit of text.match(UNIT) ?? []) {
            counts.set(unit, (counts.get(unit) ?? 0) + 1);
        }
        const ordered = [...counts.entries()]
            .filter(([, count]) => count >= minFrequency)
            .sort((first, second) => second[1] - first[1] || first[0].localeCompare(second[0]))
            .map(([unit]) => unit);
        this.#tokens = ["<unk>", ...ordered];
        this.#index = new Map(this.#tokens.map((token, index) => [token, index]));
        return this;
    }

    encode(text: string): number[] {
        return (text.match(UNIT) ?? []).map((unit) => this.#index.get(unit) ?? 0);
    }

    decode(tokens: readonly number[]): string {
        let text = "";
        for (const token of tokens) {
            if (!Number.isInteger(token) || token < 0 || token >= this.#tokens.length) {
                throw new RangeError(`token fuera de vocabulario: ${token}`);
            }
            text += token === 0 ? "\uFFFD" : (this.#tokens[token] as string);
        }
        return text;
    }
}

const UNIT = /\p{L}+|\p{N}+|[^\p{L}\p{N}\s]+|\s+/gu;

export interface WordPieceOptions {
    vocabularySize?: number;
    minFrequency?: number;
}

interface Piece {
    text: string;
    continuation: boolean;
    bytes?: Uint8Array;
}

const unitLike = (value: string): boolean => /^[\p{L}\p{N}]+$/u.test(value);

/**
 * WordPiece lossless: vocabulario de caracteres y continuaciones `##`, merges por ganancia de
 * verosimilitud y fallback de bytes para lo no visto.
 */
export class WordPieceTokenizer {
    readonly #pieces: Piece[] = [];
    readonly #index = new Map<string, number>();

    constructor() {}

    get vocabularySize(): number {
        return this.#pieces.length;
    }

    train(text: string, options: WordPieceOptions = {}): void {
        const target = options.vocabularySize ?? 2048;
        const minFrequency = options.minFrequency ?? 2;
        if (!Number.isInteger(target) || target < 256) {
            throw new RangeError(`el vocabulario debe ser al menos 256, no ${target}`);
        }
        if (!Number.isInteger(minFrequency) || minFrequency < 1) {
            throw new RangeError(`la frecuencia mínima debe ser positiva, no ${minFrequency}`);
        }
        this.#pieces.length = 0;
        this.#index.clear();
        const units = text.match(UNIT) ?? [];
        for (const unit of units) {
            if (!unitLike(unit)) {
                this.#add(unit, false);
            } else {
                const characters = [...unit];
                for (let index = 0; index < characters.length; index++) {
                    this.#add(characters[index] as string, index > 0);
                }
            }
        }
        const sequences: number[][] = [];
        for (const unit of units) {
            if (unitLike(unit)) {
                const characters = [...unit];
                sequences.push(
                    characters.map((character, index) =>
                        this.#index.get(`${index > 0 ? "##" : ""}${character}`) as number,
                    ),
                );
            }
        }

        let active = true;
        while (this.#pieces.length < target && active) {
            const counts = new Map<string, number>();
            const leftCounts = new Map<number, number>();
            const rightCounts = new Map<number, number>();
            for (const sequence of sequences) {
                for (let index = 0; index + 1 < sequence.length; index++) {
                    const left = sequence[index] as number;
                    const right = sequence[index + 1] as number;
                    const leftContinuation = this.#pieces[left]?.continuation ?? false;
                    const rightContinuation = this.#pieces[right]?.continuation ?? false;
                    const allowed =
                        !(leftContinuation === false && rightContinuation === false) &&
                        !(leftContinuation && rightContinuation === false);
                    if (allowed) {
                        const key = `${left},${right}`;
                        counts.set(key, (counts.get(key) ?? 0) + 1);
                        leftCounts.set(left, (leftCounts.get(left) ?? 0) + 1);
                        rightCounts.set(right, (rightCounts.get(right) ?? 0) + 1);
                    }
                }
            }
            let bestKey: string | null = null;
            let bestScore = 0;
            for (const [key, count] of counts) {
                if (count >= minFrequency) {
                    const [left, right] = key.split(",").map(Number) as [number, number];
                    const score = count / ((leftCounts.get(left) ?? 1) * (rightCounts.get(right) ?? 1));
                    if (score > bestScore || (score === bestScore && (bestKey === null || key < bestKey))) {
                        bestKey = key;
                        bestScore = score;
                    }
                }
            }
            if (bestKey === null) {
                active = false;
            } else {
                const [left, right] = bestKey.split(",").map(Number) as [number, number];
                const leftPiece = this.#pieces[left] as Piece;
                const rightPiece = this.#pieces[right] as Piece;
                const leftText = leftPiece.continuation ? leftPiece.text.slice(2) : leftPiece.text;
                const mergedText = `${leftText}${rightPiece.text.slice(2)}`;
                const token = this.#add(mergedText, leftPiece.continuation);
                const merged: number[][] = [];
                for (const sequence of sequences) {
                    const next: number[] = [];
                    for (let index = 0; index < sequence.length; ) {
                        if (
                            index + 1 < sequence.length &&
                            sequence[index] === left &&
                            sequence[index + 1] === right
                        ) {
                            next.push(token);
                            index += 2;
                        } else {
                            next.push(sequence[index] as number);
                            index += 1;
                        }
                    }
                    merged.push(next);
                }
                sequences.length = 0;
                sequences.push(...merged);
            }
        }
        for (let byte = 0; byte < 256; byte++) {
            this.#pieces.push({ text: "", continuation: false, bytes: Uint8Array.from([byte]) });
        }
    }

    #add(text: string, continuation: boolean): number {
        const key = `${continuation ? "##" : ""}${text}`;
        const existing = this.#index.get(key);
        if (existing !== undefined) return existing;
        const id = this.#pieces.length;
        this.#pieces.push({ text: key, continuation });
        this.#index.set(key, id);
        return id;
    }

    encode(text: string): number[] {
        const output: number[] = [];
        for (const unit of text.match(UNIT) ?? []) {
            if (!unitLike(unit)) {
                const whole = this.#index.get(unit);
                if (whole !== undefined) {
                    output.push(whole);
                } else {
                    output.push(...this.#fallback(unit));
                }
            } else {
                const characters = [...unit];
                let index = 0;
                let failed = false;
                const unitOutput: number[] = [];
                while (index < characters.length && !failed) {
                    let match: number | null = null;
                    let end = characters.length;
                    while (end > index && match === null) {
                        const substring = characters.slice(index, end).join("");
                        const key = `${index === 0 ? "" : "##"}${substring}`;
                        const token = this.#index.get(key);
                        if (token !== undefined) {
                            match = token;
                            index = end;
                        } else {
                            end -= 1;
                        }
                    }
                    if (match === null) failed = true;
                    else unitOutput.push(match);
                }
                if (failed) output.push(...this.#fallback(unit));
                else output.push(...unitOutput);
            }
        }
        return output;
    }

    #fallback(text: string): number[] {
        const bytes = new TextEncoder().encode(text);
        const start = this.#pieces.length - 256;
        return [...bytes].map((byte) => start + byte);
    }

    decode(tokens: readonly number[]): string {
        const bytes: number[] = [];
        for (const token of tokens) {
            const piece = this.#pieces[token];
            if (!piece) throw new RangeError(`token fuera de vocabulario: ${token}`);
            if (piece.bytes) {
                bytes.push(...piece.bytes);
            } else {
                const text = piece.continuation ? piece.text.slice(2) : piece.text;
                bytes.push(...new TextEncoder().encode(text));
            }
        }
        return new TextDecoder().decode(Uint8Array.from(bytes));
    }
}


// Teoría: docs/teoria/evolucion/haea.md

import { createRng } from "../core/random/index.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";
import { tournament } from "./selection.ts";
import type { Fitness, Genome, Individual, Variation } from "./types.ts";

export interface HaeaOptions {
    populationSize: number;
    generations: number;
    operators: readonly Variation[];
    tournamentSize?: number;
    elitism?: number;
    seed?: number;
    adaptive?: boolean;
    init?: (rng: RandomNumberGenerator) => Genome;
    onGeneration?: (generation: number, best: Individual) => void;
}

export interface HaeaResult {
    best: Individual;
    history: number[];
    evaluations: number;
}

const cloneGenome = (genome: Genome): Genome => new Float32Array(genome);

function selectOperator(rates: readonly number[], rng: RandomNumberGenerator): number {
    let threshold = rng.next();
    for (let index = 0; index < rates.length; index++) {
        threshold -= rates[index] as number;
        if (threshold <= 0) return index;
    }
    return rates.length - 1;
}

function adjust(
    rates: number[],
    index: number,
    rng: RandomNumberGenerator,
    reward: boolean,
): void {
    const factor = reward ? 1 + rng.next() : 1 - rng.next();
    rates[index] = Math.max(1e-6, (rates[index] as number) * factor);
    let total = 0;
    for (const rate of rates) total += rate;
    for (let position = 0; position < rates.length; position++) {
        rates[position] = (rates[position] as number) / total;
    }
}

/**
 * HAEA simplificado: población real, torneo, operadores con probabilidades por individuo y
 * premio/castigo al operador elegido según mejore o no el descendiente.
 */
export function haea(fitness: Fitness, genomeSize: number, options: HaeaOptions): HaeaResult {
    if (!Number.isInteger(genomeSize) || genomeSize <= 0) {
        throw new RangeError(`el genoma debe tener un tamaño positivo, no ${genomeSize}`);
    }
    if (!Number.isInteger(options.populationSize) || options.populationSize < 2) {
        throw new RangeError("la población necesita al menos dos individuos");
    }
    if (!Number.isInteger(options.generations) || options.generations < 0) {
        throw new RangeError("las generaciones deben ser un entero no negativo");
    }
    if (options.operators.length === 0) throw new Error("hace falta al menos un operador");
    for (const operator of options.operators) {
        if (!Number.isInteger(operator.arity) || operator.arity < 1) {
            throw new Error(`el operador «${operator.name}» tiene aridad inválida`);
        }
    }
    const tournamentSize = options.tournamentSize ?? 3;
    if (!Number.isInteger(tournamentSize) || tournamentSize < 1) {
        throw new RangeError("el torneo debe tener al menos un individuo");
    }
    const elitism = options.elitism ?? 0;
    if (!Number.isInteger(elitism) || elitism < 0 || elitism > options.populationSize) {
        throw new RangeError("el elitismo debe caber en la población");
    }
    const adaptive = options.adaptive ?? true;
    const rng = createRng(options.seed ?? 1);
    const init =
        options.init ??
        ((random: RandomNumberGenerator): Genome => {
            const genome = new Float32Array(genomeSize);
            for (let index = 0; index < genomeSize; index++) {
                genome[index] = random.next() * 2 - 1;
            }
            return genome;
        });

    const population: Individual[] = [];
    let evaluations = 0;
    for (let index = 0; index < options.populationSize; index++) {
        const genome = init(rng);
        if (genome.length !== genomeSize) {
            throw new Error(`init devolvió un genoma de ${genome.length}, no ${genomeSize}`);
        }
        population.push({ genome, fitness: fitness(genome) });
        evaluations += 1;
    }

    const rates = population.map(() =>
        new Array<number>(options.operators.length).fill(1 / options.operators.length),
    );
    let best = population[0] as Individual;
    for (const individual of population) {
        if (individual.fitness < best.fitness) best = individual;
    }
    const history = [best.fitness];

    for (let generation = 0; generation < options.generations; generation++) {
        const elites =
            elitism > 0
                ? [...population]
                      .sort((left, right) => left.fitness - right.fitness)
                      .slice(0, elitism)
                      .map((individual) => ({
                          genome: cloneGenome(individual.genome),
                          fitness: individual.fitness,
                      }))
                : [];

        for (let index = 0; index < population.length; index++) {
            const individual = population[index] as Individual;
            const operatorIndex = selectOperator(rates[index] as number[], rng);
            const operator = options.operators[operatorIndex] as Variation;
            const parents: Genome[] = [individual.genome];
            for (let parent = 1; parent < operator.arity; parent++) {
                parents.push(tournament(population, tournamentSize, rng).genome);
            }
            const child = operator.apply(parents, rng);
            const childFitness = fitness(child);
            evaluations += 1;
            if (childFitness < individual.fitness) {
                population[index] = { genome: child, fitness: childFitness };
                if (adaptive) adjust(rates[index] as number[], operatorIndex, rng, true);
            } else if (adaptive) {
                adjust(rates[index] as number[], operatorIndex, rng, false);
            }
        }

        for (const elite of elites) {
            let worstIndex = 0;
            for (let index = 1; index < population.length; index++) {
                if (
                    (population[index] as Individual).fitness >
                    (population[worstIndex] as Individual).fitness
                ) {
                    worstIndex = index;
                }
            }
            if (elite.fitness < (population[worstIndex] as Individual).fitness) {
                population[worstIndex] = elite;
            }
        }

        let generationBest = population[0] as Individual;
        for (const individual of population) {
            if (individual.fitness < generationBest.fitness) generationBest = individual;
        }
        best = generationBest;
        history.push(best.fitness);
        options.onGeneration?.(generation, best);
    }

    return {
        best: { genome: cloneGenome(best.genome), fitness: best.fitness },
        history,
        evaluations,
    };
}

/** Normal estándar por Box-Muller, a partir de dos uniformes. */
export function standardNormal(rng: RandomNumberGenerator): number {
    const first = 1 - rng.next();
    const second = rng.next();
    return Math.sqrt(-2 * Math.log(first)) * Math.cos(2 * Math.PI * second);
}

export interface GaussianMutationOptions {
    sigma?: number;
}

export class GaussianMutation implements Variation {
    readonly name = "mutacion-gaussiana";
    readonly arity = 1;
    readonly #sigma: number;

    constructor(options: GaussianMutationOptions = {}) {
        const sigma = options.sigma ?? 0.5;
        if (!Number.isFinite(sigma) || sigma <= 0) {
            throw new RangeError(`sigma debe ser positivo, no ${sigma}`);
        }
        this.#sigma = sigma;
    }

    apply(parents: readonly Genome[], rng: RandomNumberGenerator): Genome {
        const parent = parents[0];
        if (!parent) throw new Error("la mutación necesita un padre");
        const child = new Float32Array(parent.length);
        for (let index = 0; index < parent.length; index++) {
            child[index] = (parent[index] as number) + this.#sigma * standardNormal(rng);
        }
        return child;
    }
}

function twoParents(parents: readonly Genome[]): [Genome, Genome] {
    const first = parents[0];
    const second = parents[1];
    if (!first || !second) throw new Error("el cruce necesita dos padres");
    if (first.length !== second.length) throw new Error("los padres deben tener el mismo tamaño");
    return [first, second];
}

export class UniformCrossover implements Variation {
    readonly name = "cruce-uniforme";
    readonly arity = 2;

    apply(parents: readonly Genome[], rng: RandomNumberGenerator): Genome {
        const [first, second] = twoParents(parents);
        const child = new Float32Array(first.length);
        for (let index = 0; index < child.length; index++) {
            child[index] = rng.next() < 0.5 ? (first[index] as number) : (second[index] as number);
        }
        return child;
    }
}

export class ArithmeticCrossover implements Variation {
    readonly name = "cruce-aritmetico";
    readonly arity = 2;

    apply(parents: readonly Genome[], rng: RandomNumberGenerator): Genome {
        const [first, second] = twoParents(parents);
        const alpha = rng.next();
        const child = new Float32Array(first.length);
        for (let index = 0; index < child.length; index++) {
            child[index] = alpha * (first[index] as number) + (1 - alpha) * (second[index] as number);
        }
        return child;
    }
}

/** Torneo con reemplazo: elige `size` individuos al azar y devuelve el de menor fitness. */
export function tournament(
    population: readonly Individual[],
    size: number,
    rng: RandomNumberGenerator,
): Individual {
    if (population.length === 0) throw new Error("no se puede elegir de una población vacía");
    if (!Number.isInteger(size) || size <= 0) {
        throw new RangeError(`el tamaño del torneo debe ser un entero positivo, no ${size}`);
    }
    let best = population[Math.floor(rng.next() * population.length)] as Individual;
    for (let pick = 1; pick < size; pick++) {
        const candidate = population[Math.floor(rng.next() * population.length)] as Individual;
        if (candidate.fitness < best.fitness) best = candidate;
    }
    return best;
}

import { readFileSync, renameSync, writeFileSync } from "node:fs";

import { Tensor } from "../core/tensor.ts";

export const CHECKPOINT_FORMAT = "ailib-ts.checkpoint";
export const CHECKPOINT_VERSION = 1;

export interface CheckpointMetadata {
    [key: string]: unknown;
}

export interface Checkpoint {
    metadata: CheckpointMetadata;
    parameters: Record<string, Tensor>;
}

interface ParameterRecord {
    name: string;
    shape: number[];
    data: number[];
}

const isRecord = (value: unknown): value is Record<string, unknown> =>
    typeof value === "object" && value !== null && !Array.isArray(value);

const shapeSize = (shape: readonly number[]): number =>
    shape.reduce((total, dimension) => total * dimension, 1);

/** Guarda parámetros con nombre y metadatos; escribe a un temporal y renombra. */
export function saveCheckpoint(
    path: string,
    parameters: Record<string, Tensor>,
    metadata: CheckpointMetadata = {},
): void {
    const records: ParameterRecord[] = Object.entries(parameters).map(([name, tensor]) => {
        if (name.length === 0) throw new Error("un parámetro no puede llamarse vacío");
        for (const value of tensor.data) {
            if (!Number.isFinite(value)) {
                throw new Error(`el parámetro «${name}» tiene un valor no finito`);
            }
        }
        return { name, shape: [...tensor.shape], data: [...tensor.data] };
    });
    const payload = {
        format: CHECKPOINT_FORMAT,
        version: CHECKPOINT_VERSION,
        metadata,
        parameters: records,
    };
    const temporary = `${path}.tmp`;
    writeFileSync(temporary, `${JSON.stringify(payload, null, 2)}\n`, "utf8");
    renameSync(temporary, path);
}

/** Carga un checkpoint validando formato, formas y datos. */
export function loadCheckpoint(
    path: string,
    options: { requiresGrad?: boolean } = {},
): Checkpoint {
    let raw: string;
    try {
        raw = readFileSync(path, "utf8");
    } catch {
        throw new Error(`no se puede leer el checkpoint ${path}`);
    }
    let parsed: unknown;
    try {
        parsed = JSON.parse(raw);
    } catch {
        throw new Error(`${path} no es JSON válido`);
    }
    if (!isRecord(parsed)) throw new Error(`${path} no es un checkpoint válido`);
    if (parsed.format !== CHECKPOINT_FORMAT || parsed.version !== CHECKPOINT_VERSION) {
        throw new Error(`${path} no es un checkpoint ${CHECKPOINT_FORMAT} v${CHECKPOINT_VERSION}`);
    }
    if (!isRecord(parsed.metadata)) throw new Error(`${path}: metadatos inválidos`);
    if (!Array.isArray(parsed.parameters)) throw new Error(`${path}: falta la lista de parámetros`);

    const parameters: Record<string, Tensor> = {};
    for (const entry of parsed.parameters) {
        if (!isRecord(entry)) throw new Error(`${path}: parámetro inválido`);
        const name = entry.name;
        const shape = entry.shape;
        const data = entry.data;
        if (typeof name !== "string" || name.length === 0) {
            throw new Error(`${path}: nombre de parámetro inválido`);
        }
        if (name in parameters) throw new Error(`${path}: parámetro duplicado «${name}»`);
        if (
            !Array.isArray(shape) ||
            !shape.every((dimension) => Number.isInteger(dimension) && (dimension as number) >= 0)
        ) {
            throw new Error(`${path}: forma inválida en «${name}»`);
        }
        if (
            !Array.isArray(data) ||
            !data.every((value) => typeof value === "number" && Number.isFinite(value))
        ) {
            throw new Error(`${path}: datos inválidos en «${name}»`);
        }
        const dimensions = shape as number[];
        if (data.length !== shapeSize(dimensions)) {
            throw new Error(
                `${path}: «${name}» tiene ${data.length} valores y la forma pide ${shapeSize(dimensions)}`,
            );
        }
        parameters[name] = new Tensor(Float32Array.from(data as number[]), dimensions, {
            requiresGrad: options.requiresGrad ?? false,
        });
    }
    return { metadata: parsed.metadata, parameters };
}

import { createRng } from "../core/random/index.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";
import { Tensor } from "../core/tensor.ts";
import { add, mul } from "../core/tensor-ops.ts";
import { Embedding } from "../nn/embedding.ts";
import { GRU } from "../nn/gru.ts";
import { Linear } from "../nn/linear.ts";
import { crossEntropy } from "../nn/losses.ts";
import { LSTM } from "../nn/lstm.ts";
import type { CellType, RecurrentCell, RecurrentState } from "../nn/recurrent.ts";
import { RNN } from "../nn/rnn.ts";
import { sampleFromValues } from "../nn/sampling.ts";

export interface CharLMOptions {
    embeddingSize?: number;
    hiddenSize?: number;
    seed?: number;
    cell?: CellType;
}

export interface GenerateOptions {
    temperature?: number;
    topK?: number;
    topP?: number;
    seed?: number;
    rng?: RandomNumberGenerator;
}

export interface CharLMForward {
    logits: Tensor[];
    states: Tensor[];
}

export interface CharLMLoss {
    loss: Tensor;
    states: Tensor[];
}

/** LM de caracteres byte-level: Embedding → RNN → Linear sobre el vocabulario. */
export class CharLM {
    readonly embedding: Embedding;
    readonly cell: RecurrentCell;
    readonly output: Linear;

    constructor(vocabularySize: number, options: CharLMOptions = {}) {
        const embeddingSize = options.embeddingSize ?? 16;
        const hiddenSize = options.hiddenSize ?? 32;
        const seed = options.seed ?? 0;
        const cellType = options.cell ?? "rnn";
        this.embedding = new Embedding(vocabularySize, embeddingSize, { seed });
        this.cell =
            cellType === "gru"
                ? new GRU(embeddingSize, hiddenSize, { seed: seed + 1 })
                : cellType === "lstm"
                  ? new LSTM(embeddingSize, hiddenSize, { seed: seed + 1 })
                  : new RNN(embeddingSize, hiddenSize, { seed: seed + 1 });
        this.output = new Linear(hiddenSize, vocabularySize, { seed: seed + 2 });
    }

    forward(tokens: readonly number[]): CharLMForward {
        if (tokens.length === 0) throw new Error("hace falta al menos un token");
        let state: RecurrentState = this.cell.initial();
        const states: Tensor[] = [];
        for (const token of tokens) {
            state = this.cell.advance(this.embedding.row(token), state);
            states.push(state.hidden);
        }
        const logits = states.map((hidden) => this.output.forward(hidden));
        return { logits, states };
    }

    /** Pérdida media de predecir cada token a partir del anterior. */
    loss(tokens: readonly number[]): Tensor {
        return this.lossAndStates(tokens).loss;
    }

    /** Igual que `loss`, pero devuelve también los estados para estudiar el gradiente. */
    lossAndStates(tokens: readonly number[]): CharLMLoss {
        if (tokens.length < 2) {
            throw new Error("hacen falta al menos dos tokens para predecir");
        }
        const { logits, states } = this.forward(tokens.slice(0, -1));
        let total = crossEntropy(logits[0] as Tensor, [tokens[1] as number]);
        for (let index = 1; index < logits.length; index++) {
            total = add(
                total,
                crossEntropy(logits[index] as Tensor, [tokens[index + 1] as number]),
            );
        }
        return {
            loss: mul(total, Tensor.from([1 / logits.length], [])),
            states,
        };
    }

    /** Devuelve la continuación generada, sin incluir el arranque. */
    generate(start: readonly number[], length: number, options: GenerateOptions = {}): number[] {
        if (start.length === 0) throw new Error("hace falta un token de arranque");
        if (!Number.isInteger(length) || length < 0) {
            throw new RangeError(`la longitud debe ser un entero no negativo, no ${length}`);
        }
        const temperature = options.temperature ?? 0.8;
        if (!Number.isFinite(temperature) || temperature <= 0) {
            throw new RangeError(`la temperatura debe ser positiva, no ${temperature}`);
        }
        const rng = options.rng ?? createRng(options.seed ?? 0);
        let state: RecurrentState = this.cell.initial();
        for (const token of start) {
            state = this.cell.advance(this.embedding.row(token), state);
        }
        const generated: number[] = [];
        const sampling = {
            temperature,
            ...(options.topK === undefined ? {} : { topK: options.topK }),
            ...(options.topP === undefined ? {} : { topP: options.topP }),
        };
        for (let step = 0; step < length; step++) {
            const logits = this.output.forward(state.hidden);
            const next = sampleFromValues(logits.data, sampling, rng);
            generated.push(next);
            state = this.cell.advance(this.embedding.row(next), state);
        }
        return generated;
    }

    parameters(): readonly Tensor[] {
        return [
            ...this.embedding.parameters(),
            ...this.cell.parameters(),
            ...this.output.parameters(),
        ];
    }
}


// Teoría: docs/teoria/secuencias/lm-transformer.md

import { createRng } from "../core/random/index.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";
import { Tensor } from "../core/tensor.ts";
import { Embedding } from "../nn/embedding.ts";
import { KVCache } from "../nn/kv-cache.ts";
import type { AttentionCache } from "../nn/kv-cache.ts";
import { LayerNorm } from "../nn/layernorm.ts";
import { Linear } from "../nn/linear.ts";
import { crossEntropy } from "../nn/losses.ts";
import { RMSNorm } from "../nn/rmsnorm.ts";
import { sampleFromValues } from "../nn/sampling.ts";
import { TransformerBlock } from "../nn/transformer-block.ts";

export interface TransformerLMOptions {
    modelSize?: number;
    heads?: number;
    blocks?: number;
    hiddenSize?: number;
    seed?: number;
    rope?: boolean;
    attention?: boolean;
    ffn?: boolean;
    normKind?: "layer" | "rms";
    ffnKind?: "relu" | "swiglu";
    parallel?: boolean;
}

export interface TransformerGenerateOptions {
    temperature?: number;
    topK?: number;
    topP?: number;
    seed?: number;
    rng?: RandomNumberGenerator;
    useCache?: boolean;
}

/** LM byte-level con bloques transformer, RoPE y máscara causal. */
export class TransformerLM {
    readonly embedding: Embedding;
    readonly blocks: readonly TransformerBlock[];
    readonly finalNorm: LayerNorm | RMSNorm;
    readonly output: Linear;
    readonly #vocabularySize: number;
    readonly #modelSize: number;
    readonly #heads: number;
    readonly #hiddenSize: number;

    constructor(vocabularySize: number, options: TransformerLMOptions = {}) {
        const modelSize = options.modelSize ?? 32;
        const heads = options.heads ?? 4;
        const blocks = options.blocks ?? 1;
        const hiddenSize = options.hiddenSize ?? modelSize * 4;
        const seed = options.seed ?? 0;
        if (!Number.isInteger(blocks) || blocks <= 0) {
            throw new Error(`el transformer necesita al menos un bloque, no ${blocks}`);
        }
        this.#vocabularySize = vocabularySize;
        this.#modelSize = modelSize;
        this.#heads = heads;
        this.#hiddenSize = hiddenSize;
        this.embedding = new Embedding(vocabularySize, modelSize, { seed });
        this.blocks = Array.from(
            { length: blocks },
            (_, index) =>
                new TransformerBlock(modelSize, heads, {
                    hiddenSize,
                    seed: seed + 10 + index * 10,
                    rope: options.rope ?? true,
                    attention: options.attention ?? true,
                    ffn: options.ffn ?? true,
                    normKind: options.normKind ?? "layer",
                    ffnKind: options.ffnKind ?? "relu",
                    parallel: options.parallel ?? false,
                }),
        );
        this.finalNorm =
            (options.normKind ?? "layer") === "rms"
                ? new RMSNorm(modelSize)
                : new LayerNorm(modelSize);
        this.output = new Linear(modelSize, vocabularySize, { seed: seed + 10 + blocks * 10 });
    }

    forward(tokens: readonly number[]): Tensor {
        if (tokens.length === 0) throw new Error("hace falta al menos un token");
        let value = this.embedding.forward(tokens);
        for (const block of this.blocks) {
            value = block.forward(value).output;
        }
        return this.output.forward(this.finalNorm.forward(value));
    }

    /** Cross-entropy media de predecir cada token a partir del anterior. */
    loss(tokens: readonly number[]): Tensor {
        if (tokens.length < 2) {
            throw new Error("hacen falta al menos dos tokens para predecir");
        }
        return crossEntropy(this.forward(tokens.slice(0, -1)), tokens.slice(1));
    }

    /** Devuelve la continuación generada, sin incluir el arranque. */
    generate(
        start: readonly number[],
        length: number,
        options: TransformerGenerateOptions = {},
    ): number[] {
        if (start.length === 0) throw new Error("hace falta un token de arranque");
        if (!Number.isInteger(length) || length < 0) {
            throw new RangeError(`la longitud debe ser un entero no negativo, no ${length}`);
        }
        const temperature = options.temperature ?? 0.8;
        if (!Number.isFinite(temperature) || temperature <= 0) {
            throw new RangeError(`la temperatura debe ser positiva, no ${temperature}`);
        }
        const rng = options.rng ?? createRng(options.seed ?? 0);
        const sampling: { temperature: number; topK?: number; topP?: number } = {
            temperature,
            ...(options.topK === undefined ? {} : { topK: options.topK }),
            ...(options.topP === undefined ? {} : { topP: options.topP }),
        };
        if (options.useCache) return this.#generateCached(start, length, sampling, rng);
        const sequence = [...start];
        const generated: number[] = [];
        for (let step = 0; step < length; step++) {
            const logits = this.forward(sequence);
            const row = sequence.length - 1;
            const values = logits.data.subarray(
                row * this.#vocabularySize,
                (row + 1) * this.#vocabularySize,
            );
            const next = sampleFromValues(values, sampling, rng);
            generated.push(next);
            sequence.push(next);
        }
        return generated;
    }

    #generateCached(
        start: readonly number[],
        length: number,
        sampling: { temperature: number; topK?: number; topP?: number },
        rng: RandomNumberGenerator,
    ): number[] {
        if (length === 0) return [];
        const cache = new KVCache(this.blocks.length, this.#heads, this.#modelSize / this.#heads);
        let value = this.embedding.row(start[0] as number);
        for (let block = 0; block < this.blocks.length; block++) {
            value = (this.blocks[block] as TransformerBlock).forwardCached(
                value,
                cache.layers[block] as AttentionCache,
                0,
            );
        }
        for (let position = 1; position < start.length; position++) {
            value = this.embedding.row(start[position] as number);
            for (let block = 0; block < this.blocks.length; block++) {
                value = (this.blocks[block] as TransformerBlock).forwardCached(
                    value,
                    cache.layers[block] as AttentionCache,
                    position,
                );
            }
        }

        const generated: number[] = [];
        for (let step = 0; step < length; step++) {
            const logits = this.output.forward(this.finalNorm.forward(value));
            const next = sampleFromValues(logits.data, sampling, rng);
            generated.push(next);
            if (step + 1 < length) {
                const position = start.length + step;
                value = this.embedding.row(next);
                for (let block = 0; block < this.blocks.length; block++) {
                    value = (this.blocks[block] as TransformerBlock).forwardCached(
                        value,
                        cache.layers[block] as AttentionCache,
                        position,
                    );
                }
            }
        }
        return generated;
    }

    parameters(): readonly Tensor[] {
        return [
            ...this.embedding.parameters(),
            ...this.blocks.flatMap((block) => block.parameters()),
            ...this.finalNorm.parameters(),
            ...this.output.parameters(),
        ];
    }

    get architecture(): {
        modelSize: number;
        heads: number;
        blocks: number;
        hiddenSize: number;
        vocabularySize: number;
    } {
        return {
            modelSize: this.#modelSize,
            heads: this.#heads,
            blocks: this.blocks.length,
            hiddenSize: this.#hiddenSize,
            vocabularySize: this.#vocabularySize,
        };
    }

    parameterCount(): number {
        return this.parameters().reduce((total, parameter) => total + parameter.size, 0);
    }

    /** Nombres estables para guardar y reanudar checkpoints. */
    namedParameters(): Record<string, Tensor> {
        const named: Record<string, Tensor> = {
            "embedding.weights": this.embedding.weights,
        };
        this.blocks.forEach((block, blockIndex) => {
            block.parameters().forEach((parameter, position) => {
                named[`block.${blockIndex}.${position}`] = parameter;
            });
        });
        named["finalNorm.gamma"] = this.finalNorm.gamma;
        if ("beta" in this.finalNorm) named["finalNorm.beta"] = this.finalNorm.beta;
        named["output.weights"] = this.output.weights;
        named["output.bias"] = this.output.bias;
        return named;
    }

    /** Copia parámetros por nombre; exige el conjunto completo y formas compatibles. */
    loadParameters(parameters: Record<string, Tensor>): void {
        const named = this.namedParameters();
        const expected = Object.keys(named).sort();
        const actual = Object.keys(parameters).sort();
        if (expected.join("\u0000") !== actual.join("\u0000")) {
            throw new Error("el checkpoint no coincide con la arquitectura del modelo");
        }
        for (const name of expected) {
            const source = parameters[name] as Tensor;
            const target = named[name] as Tensor;
            if (source.size !== target.size) {
                throw new Error(`«${name}» tiene ${source.size} valores, se esperaban ${target.size}`);
            }
            target.data.set(source.data);
        }
    }
}


// Teoría: docs/teoria/secuencias/atencion.md

import { Tensor } from "../core/tensor.ts";

export interface AttentionOptions {
    causal?: boolean;
}

export interface AttentionResult {
    output: Tensor;
    weights: Tensor;
}

/**
 * Atención escalada de un solo ejemplo: scores = QK^T / sqrt(d), softmax por filas y salida P·V.
 * Con `causal: true`, cada posición solo puede mirar hacia atrás.
 */
export function scaledDotProductAttention(
    queries: Tensor,
    keys: Tensor,
    values: Tensor,
    options: AttentionOptions = {},
): AttentionResult {
    if (queries.rank !== 2 || keys.rank !== 2 || values.rank !== 2) {
        throw new Error("Q, K y V deben ser matrices [secuencia, dimensión]");
    }
    const sequence = queries.shape[0] as number;
    const dimension = queries.shape[1] as number;
    const keySequence = keys.shape[0] as number;
    const valueDimension = values.shape[1] as number;
    if (sequence === 0 || dimension === 0 || keySequence === 0 || valueDimension === 0) {
        throw new Error("la atención necesita dimensiones positivas");
    }
    if ((keys.shape[1] as number) !== dimension) {
        throw new Error(
            `K debe ser [${keySequence}, ${dimension}], no ${JSON.stringify(keys.shape)}`,
        );
    }
    if ((values.shape[0] as number) !== keySequence) {
        throw new Error(`V debe tener ${keySequence} filas, no ${values.shape[0]}`);
    }

    const causal = options.causal ?? true;
    const scale = 1 / Math.sqrt(dimension);
    const scores = new Float32Array(sequence * keySequence);
    for (let row = 0; row < sequence; row++) {
        for (let column = 0; column < keySequence; column++) {
            if (causal && column > row) {
                scores[row * keySequence + column] = -Infinity;
            } else {
                let dot = 0;
                for (let index = 0; index < dimension; index++) {
                    dot += (queries.data[row * dimension + index] as number) *
                        (keys.data[column * dimension + index] as number);
                }
                scores[row * keySequence + column] = dot * scale;
            }
        }
    }

    const weights = new Float32Array(sequence * keySequence);
    for (let row = 0; row < sequence; row++) {
        let maximum = -Infinity;
        for (let column = 0; column < keySequence; column++) {
            maximum = Math.max(maximum, scores[row * keySequence + column] as number);
        }
        let total = 0;
        for (let column = 0; column < keySequence; column++) {
            const value = Math.exp((scores[row * keySequence + column] as number) - maximum);
            weights[row * keySequence + column] = value;
            total += value;
        }
        for (let column = 0; column < keySequence; column++) {
            weights[row * keySequence + column] =
                (weights[row * keySequence + column] as number) / total;
        }
    }

    const outputData = new Float32Array(sequence * valueDimension);
    for (let row = 0; row < sequence; row++) {
        for (let index = 0; index < valueDimension; index++) {
            let total = 0;
            for (let column = 0; column < keySequence; column++) {
                total += (weights[row * keySequence + column] as number) *
                    (values.data[column * valueDimension + index] as number);
            }
            outputData[row * valueDimension + index] = total;
        }
    }
    const output = new Tensor(outputData, [sequence, valueDimension]);
    const weightTensor = new Tensor(weights, [sequence, keySequence]);
    if (queries.requiresGrad || keys.requiresGrad || values.requiresGrad) {
        output.requiresGrad = true;
        output.setGraph([queries, keys, values], (gradient) => {
            const dV = new Float32Array(keySequence * valueDimension);
            for (let row = 0; row < sequence; row++) {
                for (let column = 0; column < keySequence; column++) {
                    const weight = weights[row * keySequence + column] as number;
                    for (let index = 0; index < valueDimension; index++) {
                        dV[column * valueDimension + index] =
                            (dV[column * valueDimension + index] as number) +
                            weight * (gradient.data[row * valueDimension + index] as number);
                    }
                }
            }

            const dP = new Float32Array(sequence * keySequence);
            for (let row = 0; row < sequence; row++) {
                for (let column = 0; column < keySequence; column++) {
                    let total = 0;
                    for (let index = 0; index < valueDimension; index++) {
                        total += (gradient.data[row * valueDimension + index] as number) *
                            (values.data[column * valueDimension + index] as number);
                    }
                    dP[row * keySequence + column] = total;
                }
            }

            const dS = new Float32Array(sequence * keySequence);
            for (let row = 0; row < sequence; row++) {
                let rowDot = 0;
                for (let column = 0; column < keySequence; column++) {
                    rowDot += (dP[row * keySequence + column] as number) *
                        (weights[row * keySequence + column] as number);
                }
                for (let column = 0; column < keySequence; column++) {
                    if (!causal || column <= row) {
                        dS[row * keySequence + column] =
                            (weights[row * keySequence + column] as number) *
                            ((dP[row * keySequence + column] as number) - rowDot);
                    }
                }
            }

            const dQ = new Float32Array(sequence * dimension);
            const dK = new Float32Array(keySequence * dimension);
            for (let row = 0; row < sequence; row++) {
                for (let index = 0; index < dimension; index++) {
                    let total = 0;
                    for (let column = 0; column < keySequence; column++) {
                        total += (dS[row * keySequence + column] as number) *
                            (keys.data[column * dimension + index] as number);
                    }
                    dQ[row * dimension + index] = total * scale;
                }
            }
            for (let column = 0; column < keySequence; column++) {
                for (let index = 0; index < dimension; index++) {
                    let total = 0;
                    for (let row = 0; row < sequence; row++) {
                        total += (dS[row * keySequence + column] as number) *
                            (queries.data[row * dimension + index] as number);
                    }
                    dK[column * dimension + index] = total * scale;
                }
            }

            queries.addGrad(new Tensor(dQ, [sequence, dimension]));
            keys.addGrad(new Tensor(dK, [keySequence, dimension]));
            values.addGrad(new Tensor(dV, [keySequence, valueDimension]));
        });
    }
    return { output, weights: weightTensor };
}

import { createRng } from "../core/random/index.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";
import { Tensor } from "../core/tensor.ts";

export interface EmbeddingOptions {
    seed?: number;
    rng?: RandomNumberGenerator;
}

/** Tabla de embeddings: cada token se busca como una fila de `weights`. */
export class Embedding {
    readonly weights: Tensor;
    readonly #vocabularySize: number;
    readonly #dimension: number;

    constructor(vocabularySize: number, dimension: number, options: EmbeddingOptions = {}) {
        if (!Number.isInteger(vocabularySize) || vocabularySize <= 0) {
            throw new Error(`el vocabulario debe ser positivo, no ${vocabularySize}`);
        }
        if (!Number.isInteger(dimension) || dimension <= 0) {
            throw new Error(`la dimensión debe ser positiva, no ${dimension}`);
        }
        const rng = options.rng ?? createRng(options.seed ?? 0);
        const limit = 1 / Math.sqrt(dimension);
        const data = new Float32Array(vocabularySize * dimension);
        for (let index = 0; index < data.length; index++) {
            data[index] = (rng.next() * 2 - 1) * limit;
        }
        this.weights = new Tensor(data, [vocabularySize, dimension], { requiresGrad: true });
        this.#vocabularySize = vocabularySize;
        this.#dimension = dimension;
    }

    get vocabularySize(): number {
        return this.#vocabularySize;
    }

    get dimension(): number {
        return this.#dimension;
    }

    /** Devuelve una fila por índice; el backward reparte el gradiente en esas filas. */
    forward(indices: readonly number[]): Tensor {
        const rows = indices.length;
        const result = new Tensor(new Float32Array(rows * this.#dimension), [
            rows,
            this.#dimension,
        ]);
        for (let row = 0; row < rows; row++) {
            const token = this.#validate(indices[row] as number, row);
            result.data.set(
                this.weights.data.subarray(
                    token * this.#dimension,
                    (token + 1) * this.#dimension,
                ),
                row * this.#dimension,
            );
        }
        if (this.weights.requiresGrad) {
            result.requiresGrad = true;
            result.setGraph([this.weights], (gradient) => {
                const values = new Float32Array(this.weights.size);
                for (let row = 0; row < rows; row++) {
                    const token = indices[row] as number;
                    for (let column = 0; column < this.#dimension; column++) {
                        const position = token * this.#dimension + column;
                        values[position] =
                            (values[position] as number) +
                            (gradient.data[row * this.#dimension + column] as number);
                    }
                }
                this.weights.addGrad(new Tensor(values, this.weights.shape));
            });
        }
        return result;
    }

    row(index: number): Tensor {
        return this.forward([index]);
    }

    parameters(): readonly Tensor[] {
        return [this.weights];
    }

    #validate(index: number, row: number): number {
        if (!Number.isInteger(index) || index < 0 || index >= this.#vocabularySize) {
            throw new RangeError(
                `índice ${index} fuera de [0, ${this.#vocabularySize}) en la fila ${row}`,
            );
        }
        return index;
    }
}

    }

// Teoría: docs/teoria/secuencias/gru-lstm.md

import { createRng } from "../core/random/index.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";
import { Tensor } from "../core/tensor.ts";
import { add, matmul, mul, sub, tanh } from "../core/tensor-ops.ts";
import type { RecurrentCell, RecurrentState } from "./recurrent.ts";
import { sigmoidTensor } from "./sigmoid.ts";

export interface GRUOptions {
    seed?: number;
    rng?: RandomNumberGenerator;
}

function randomMatrix(
    rows: number,
    columns: number,
    rng: RandomNumberGenerator,
    limit: number,
): Tensor {
    const data = new Float32Array(rows * columns);
    for (let index = 0; index < data.length; index++) {
        data[index] = (rng.next() * 2 - 1) * limit;
    }
    return new Tensor(data, [rows, columns], { requiresGrad: true });
}

/** GRU: compuertas de actualización y reinicio sobre el estado oculto. */
export class GRU implements RecurrentCell {
    readonly #zInput: Tensor;
    readonly #zHidden: Tensor;
    readonly #zBias: Tensor;
    readonly #rInput: Tensor;
    readonly #rHidden: Tensor;
    readonly #rBias: Tensor;
    readonly #nInput: Tensor;
    readonly #nHidden: Tensor;
    readonly #nBias: Tensor;
    readonly #inputSize: number;
    readonly #hiddenSize: number;

    constructor(inputSize: number, hiddenSize: number, options: GRUOptions = {}) {
        if (!Number.isInteger(inputSize) || inputSize <= 0) {
            throw new Error(`la entrada debe ser positiva, no ${inputSize}`);
        }
        if (!Number.isInteger(hiddenSize) || hiddenSize <= 0) {
            throw new Error(`el estado oculto debe ser positivo, no ${hiddenSize}`);
        }
        const rng = options.rng ?? createRng(options.seed ?? 0);
        const inputLimit = Math.sqrt(6 / (inputSize + hiddenSize));
        const hiddenLimit = Math.sqrt(6 / (hiddenSize + hiddenSize));
        this.#zInput = randomMatrix(inputSize, hiddenSize, rng, inputLimit);
        this.#zHidden = randomMatrix(hiddenSize, hiddenSize, rng, hiddenLimit);
        this.#zBias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#rInput = randomMatrix(inputSize, hiddenSize, rng, inputLimit);
        this.#rHidden = randomMatrix(hiddenSize, hiddenSize, rng, hiddenLimit);
        this.#rBias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#nInput = randomMatrix(inputSize, hiddenSize, rng, inputLimit);
        this.#nHidden = randomMatrix(hiddenSize, hiddenSize, rng, hiddenLimit);
        this.#nBias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#inputSize = inputSize;
        this.#hiddenSize = hiddenSize;
    }

    get inputSize(): number {
        return this.#inputSize;
    }

    get hiddenSize(): number {
        return this.#hiddenSize;
    }

    initial(): RecurrentState {
        return { hidden: Tensor.zeros([1, this.#hiddenSize]) };
    }

    advance(input: Tensor, state: RecurrentState): RecurrentState {
        return { hidden: this.step(input, state.hidden) };
    }

    step(input: Tensor, hidden: Tensor): Tensor {
        if (input.rank !== 2 || input.shape[0] !== 1 || input.shape[1] !== this.#inputSize) {
            throw new Error(
                `la entrada debe ser [1, ${this.#inputSize}], no ${JSON.stringify(input.shape)}`,
            );
        }
        if (hidden.rank !== 2 || hidden.shape[0] !== 1 || hidden.shape[1] !== this.#hiddenSize) {
            throw new Error(
                `el estado debe ser [1, ${this.#hiddenSize}], no ${JSON.stringify(hidden.shape)}`,
            );
        }
        const z = sigmoidTensor(
            add(add(matmul(input, this.#zInput), matmul(hidden, this.#zHidden)), this.#zBias),
        );
        const r = sigmoidTensor(
            add(add(matmul(input, this.#rInput), matmul(hidden, this.#rHidden)), this.#rBias),
        );
        const candidate = tanh(
            add(
                add(matmul(input, this.#nInput), matmul(mul(r, hidden), this.#nHidden)),
                this.#nBias,
            ),
        );
        return add(mul(sub(Tensor.ones(z.shape), z), candidate), mul(z, hidden));
    }

    parameters(): readonly Tensor[] {
        return [
            this.#zInput,
            this.#zHidden,
            this.#zBias,
            this.#rInput,
            this.#rHidden,
            this.#rBias,
            this.#nInput,
            this.#nHidden,
            this.#nBias,
        ];
    }
}

export interface AttentionCache {
    keys: Tensor[];
    values: Tensor[];
}

function emptyRows(columns: number): Tensor {
    return new Tensor(new Float32Array(0), [0, columns]);
}

/** Añade filas a una matriz sin construir grafo: el pasado de la caché es constante. */
export function appendRows(previous: Tensor, next: Tensor): Tensor {
    if (previous.rank !== 2 || next.rank !== 2) {
        throw new Error("appendRows espera matrices");
    }
    const columns = previous.shape[1] as number;
    if ((next.shape[1] as number) !== columns) {
        throw new Error("appendRows necesita las mismas columnas");
    }
    const rows = (previous.shape[0] as number) + (next.shape[0] as number);
    const data = new Float32Array(rows * columns);
    data.set(previous.data);
    data.set(next.data, previous.size);
    return new Tensor(data, [rows, columns]);
}

/** Caché por bloque y por cabeza; crece una posición en cada paso. */
export class KVCache {
    readonly layers: AttentionCache[];

    constructor(layers: number, heads: number, headSize: number) {
        if (!Number.isInteger(layers) || layers <= 0) {
            throw new Error(`capas inválidas: ${layers}`);
        }
        if (!Number.isInteger(heads) || heads <= 0) {
            throw new Error(`cabezas inválidas: ${heads}`);
        }
        if (!Number.isInteger(headSize) || headSize <= 0) {
            throw new Error(`tamaño de cabeza inválido: ${headSize}`);
        }
        this.layers = Array.from({ length: layers }, () => ({
            keys: Array.from({ length: heads }, () => emptyRows(headSize)),
            values: Array.from({ length: heads }, () => emptyRows(headSize)),
        }));
    }

    get length(): number {
        return (this.layers[0]?.keys[0]?.shape[0] as number | undefined) ?? 0;
    }
}

import { Tensor } from "../core/tensor.ts";

export interface LayerNormOptions {
    epsilon?: number;
}

/** LayerNorm por fila: normaliza la última dimensión y aplica gamma/beta. */
export class LayerNorm {
    readonly gamma: Tensor;
    readonly beta: Tensor;
    readonly #size: number;
    readonly #epsilon: number;

    constructor(size: number, options: LayerNormOptions = {}) {
        if (!Number.isInteger(size) || size <= 0) {
            throw new Error(`LayerNorm necesita una dimensión positiva, no ${size}`);
        }
        const epsilon = options.epsilon ?? 1e-5;
        if (!Number.isFinite(epsilon) || epsilon <= 0) {
            throw new RangeError(`epsilon debe ser positivo, no ${epsilon}`);
        }
        this.gamma = Tensor.ones([size], { requiresGrad: true });
        this.beta = Tensor.zeros([size], { requiresGrad: true });
        this.#size = size;
        this.#epsilon = epsilon;
    }

    forward(input: Tensor): Tensor {
        if (input.rank !== 2 || (input.shape[1] as number) !== this.#size) {
            throw new Error(
                `LayerNorm espera [filas, ${this.#size}], no ${JSON.stringify(input.shape)}`,
            );
        }
        const rows = input.shape[0] as number;
        const size = this.#size;
        const normalized = new Float32Array(rows * size);
        const inverseDeviation = new Float32Array(rows);
        const output = new Float32Array(rows * size);
        for (let row = 0; row < rows; row++) {
            let sum = 0;
            for (let column = 0; column < size; column++) {
                sum += input.data[row * size + column] as number;
            }
            const mean = sum / size;
            let squares = 0;
            for (let column = 0; column < size; column++) {
                const delta = (input.data[row * size + column] as number) - mean;
                squares += delta * delta;
            }
            const variance = squares / size;
            const inverse = 1 / Math.sqrt(variance + this.#epsilon);
            inverseDeviation[row] = inverse;
            for (let column = 0; column < size; column++) {
                const value =
                    ((input.data[row * size + column] as number) - mean) * inverse;
                normalized[row * size + column] = value;
                output[row * size + column] =
                    value * (this.gamma.data[column] as number) +
                    (this.beta.data[column] as number);
            }
        }

        const result = new Tensor(output, [rows, size]);
        if (input.requiresGrad || this.gamma.requiresGrad || this.beta.requiresGrad) {
            result.requiresGrad = true;
            result.setGraph([input, this.gamma, this.beta], (gradient) => {
                const dGamma = new Float32Array(size);
                const dBeta = new Float32Array(size);
                const dNormalized = new Float32Array(rows * size);
                for (let row = 0; row < rows; row++) {
                    for (let column = 0; column < size; column++) {
                        const value = gradient.data[row * size + column] as number;
                        dGamma[column] = (dGamma[column] as number) +
                            value * (normalized[row * size + column] as number);
                        dBeta[column] = (dBeta[column] as number) + value;
                        dNormalized[row * size + column] =
                            value * (this.gamma.data[column] as number);
                    }
                }
                const dInput = new Float32Array(rows * size);
                for (let row = 0; row < rows; row++) {
                    let sumNormalized = 0;
                    let sumNormalizedSquared = 0;
                    for (let column = 0; column < size; column++) {
                        const value = dNormalized[row * size + column] as number;
                        sumNormalized += value;
                        sumNormalizedSquared +=
                            value * (normalized[row * size + column] as number);
                    }
                    const inverse = inverseDeviation[row] as number;
                    for (let column = 0; column < size; column++) {
                        const value = dNormalized[row * size + column] as number;
                        dInput[row * size + column] =
                            (inverse / size) *
                            (size * value - sumNormalized -
                                (normalized[row * size + column] as number) *
                                    sumNormalizedSquared);
                    }
                }
                input.addGrad(new Tensor(dInput, input.shape));
                this.gamma.addGrad(new Tensor(dGamma, [size]));
                this.beta.addGrad(new Tensor(dBeta, [size]));
            });
        }
        return result;
    }

    parameters(): readonly Tensor[] {
        return [this.gamma, this.beta];
    }
}


// Teoría: docs/teoria/redes/capa-lineal.md

import { createRng } from "../core/random/index.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";
import { Tensor } from "../core/tensor.ts";
import { add, matmul } from "../core/tensor-ops.ts";

export interface LinearOptions {
    seed?: number;
    rng?: RandomNumberGenerator;
}

/** Capa densa: y = xW + b, con W [entrada, salida] y b [salida]. */
export class Linear {
    readonly weights: Tensor;
    readonly bias: Tensor;
    readonly #inputSize: number;
    readonly #outputSize: number;

    constructor(inputSize: number, outputSize: number, options: LinearOptions = {}) {
        if (!Number.isInteger(inputSize) || inputSize <= 0 || !Number.isInteger(outputSize) || outputSize <= 0) {
            throw new Error(`tamaños de capa inválidos: ${inputSize} → ${outputSize}`);
        }
        const rng = options.rng ?? createRng(options.seed ?? 0);
        // Xavier uniforme: U(-sqrt(6/(in+out)), +sqrt(6/(in+out))).
        const limit = Math.sqrt(6 / (inputSize + outputSize));
        const data = new Float32Array(inputSize * outputSize);
        for (let index = 0; index < data.length; index++) {
            data[index] = (rng.next() * 2 - 1) * limit;
        }
        this.weights = new Tensor(data, [inputSize, outputSize], { requiresGrad: true });
        this.bias = Tensor.zeros([outputSize], { requiresGrad: true });
        this.#inputSize = inputSize;
        this.#outputSize = outputSize;
    }

    get inputSize(): number {
        return this.#inputSize;
    }

    get outputSize(): number {
        return this.#outputSize;
    }

    forward(input: Tensor): Tensor {
        if (input.rank !== 2 || (input.shape[1] as number) !== this.#inputSize) {
            throw new Error(
                `la entrada debe ser [lote, ${this.#inputSize}], no ${JSON.stringify(input.shape)}`,
            );
        }
        return add(matmul(input, this.weights), this.bias);
    }

    parameters(): readonly Tensor[] {
        return [this.weights, this.bias];
    }
}


// Teoría: docs/teoria/entrenamiento/perdidas.md

import { Tensor } from "../core/tensor.ts";
import { mean, mul, sub } from "../core/tensor-ops.ts";

/** Error cuadrático medio; predicción y objetivo deben tener la misma forma. */
export function mse(prediction: Tensor, target: Tensor): Tensor {
    const sameShape =
        prediction.shape.length === target.shape.length &&
        prediction.shape.every((dimension, axis) => dimension === target.shape[axis]);
    if (!sameShape) {
        throw new Error(
            `mse espera formas iguales, no ${JSON.stringify(prediction.shape)} y ${JSON.stringify(target.shape)}`,
        );
    }
    const difference = sub(prediction, target);
    return mean(mul(difference, difference));
}

/** Entropía cruzada multiclase: logits [lote, clases] y la clase correcta por fila. */
export function crossEntropy(logits: Tensor, targets: readonly number[]): Tensor {
    if (logits.rank !== 2) {
        throw new Error("crossEntropy espera logits de rango 2 [lote, clases]");
    }
    const batch = logits.shape[0] as number;
    const classes = logits.shape[1] as number;
    if (classes <= 0) throw new Error("crossEntropy necesita al menos una clase");
    if (batch === 0) throw new Error("crossEntropy necesita un lote no vacío");
    if (targets.length !== batch) {
        throw new Error(`se esperaban ${batch} objetivos, no ${targets.length}`);
    }
    targets.forEach((target, row) => {
        if (!Number.isInteger(target) || target < 0 || target >= classes) {
            throw new RangeError(`objetivo ${target} fuera de [0, ${classes}) en la fila ${row}`);
        }
    });

    // Forward estable: restar el máximo por fila antes de exponenciar.
    const exponentials = new Float32Array(batch * classes);
    let total = 0;
    for (let row = 0; row < batch; row++) {
        let maximum = -Infinity;
        for (let column = 0; column < classes; column++) {
            maximum = Math.max(maximum, logits.get(row, column));
        }
        let sum = 0;
        for (let column = 0; column < classes; column++) {
            const value = Math.exp(logits.get(row, column) - maximum);
            exponentials[row * classes + column] = value;
            sum += value;
        }
        total += maximum + Math.log(sum) - logits.get(row, targets[row] as number);
    }

    const result = Tensor.from([total / batch], []);
    if (logits.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([logits], (gradient) => {
            const upstream = gradient.get();
            const values = new Float32Array(batch * classes);
            for (let row = 0; row < batch; row++) {
                let sum = 0;
                for (let column = 0; column < classes; column++) {
                    sum += exponentials[row * classes + column] as number;
                }
                for (let column = 0; column < classes; column++) {
                    const probability = (exponentials[row * classes + column] as number) / sum;
                    const oneHot = column === targets[row] ? 1 : 0;
                    values[row * classes + column] = ((probability - oneHot) * upstream) / batch;
                }
            }
            logits.addGrad(new Tensor(values, [batch, classes]));
        });
    }
    return result;
}

import { createRng } from "../core/random/index.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";
import { Tensor } from "../core/tensor.ts";
import { add, matmul, mul, tanh } from "../core/tensor-ops.ts";
import type { RecurrentCell, RecurrentState } from "./recurrent.ts";
import { sigmoidTensor } from "./sigmoid.ts";

export interface LSTMOptions {
    seed?: number;
    rng?: RandomNumberGenerator;
}

function randomMatrix(
    rows: number,
    columns: number,
    rng: RandomNumberGenerator,
    limit: number,
): Tensor {
    const data = new Float32Array(rows * columns);
    for (let index = 0; index < data.length; index++) {
        data[index] = (rng.next() * 2 - 1) * limit;
    }
    return new Tensor(data, [rows, columns], { requiresGrad: true });
}

/** LSTM: compuertas de entrada, olvido y salida, más un estado de celda. */
export class LSTM implements RecurrentCell {
    readonly #iInput: Tensor;
    readonly #iHidden: Tensor;
    readonly #iBias: Tensor;
    readonly #fInput: Tensor;
    readonly #fHidden: Tensor;
    readonly #fBias: Tensor;
    readonly #oInput: Tensor;
    readonly #oHidden: Tensor;
    readonly #oBias: Tensor;
    readonly #gInput: Tensor;
    readonly #gHidden: Tensor;
    readonly #gBias: Tensor;
    readonly #inputSize: number;
    readonly #hiddenSize: number;

    constructor(inputSize: number, hiddenSize: number, options: LSTMOptions = {}) {
        if (!Number.isInteger(inputSize) || inputSize <= 0) {
            throw new Error(`la entrada debe ser positiva, no ${inputSize}`);
        }
        if (!Number.isInteger(hiddenSize) || hiddenSize <= 0) {
            throw new Error(`el estado oculto debe ser positivo, no ${hiddenSize}`);
        }
        const rng = options.rng ?? createRng(options.seed ?? 0);
        const inputLimit = Math.sqrt(6 / (inputSize + hiddenSize));
        const hiddenLimit = Math.sqrt(6 / (hiddenSize + hiddenSize));
        this.#iInput = randomMatrix(inputSize, hiddenSize, rng, inputLimit);
        this.#iHidden = randomMatrix(hiddenSize, hiddenSize, rng, hiddenLimit);
        this.#iBias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#fInput = randomMatrix(inputSize, hiddenSize, rng, inputLimit);
        this.#fHidden = randomMatrix(hiddenSize, hiddenSize, rng, hiddenLimit);
        this.#fBias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#oInput = randomMatrix(inputSize, hiddenSize, rng, inputLimit);
        this.#oHidden = randomMatrix(hiddenSize, hiddenSize, rng, hiddenLimit);
        this.#oBias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#gInput = randomMatrix(inputSize, hiddenSize, rng, inputLimit);
        this.#gHidden = randomMatrix(hiddenSize, hiddenSize, rng, hiddenLimit);
        this.#gBias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#inputSize = inputSize;
        this.#hiddenSize = hiddenSize;
    }

    get inputSize(): number {
        return this.#inputSize;
    }

    get hiddenSize(): number {
        return this.#hiddenSize;
    }

    initial(): RecurrentState {
        return {
            hidden: Tensor.zeros([1, this.#hiddenSize]),
            cell: Tensor.zeros([1, this.#hiddenSize]),
        };
    }

    advance(input: Tensor, state: RecurrentState): RecurrentState {
        const cell = state.cell ?? Tensor.zeros([1, this.#hiddenSize]);
        return this.step(input, state.hidden, cell);
    }

    step(input: Tensor, hidden: Tensor, cell: Tensor): RecurrentState {
        if (input.rank !== 2 || input.shape[0] !== 1 || input.shape[1] !== this.#inputSize) {
            throw new Error(
                `la entrada debe ser [1, ${this.#inputSize}], no ${JSON.stringify(input.shape)}`,
            );
        }
        if (hidden.rank !== 2 || hidden.shape[0] !== 1 || hidden.shape[1] !== this.#hiddenSize) {
            throw new Error(
                `el estado debe ser [1, ${this.#hiddenSize}], no ${JSON.stringify(hidden.shape)}`,
            );
        }
        if (cell.rank !== 2 || cell.shape[0] !== 1 || cell.shape[1] !== this.#hiddenSize) {
            throw new Error(
                `la celda debe ser [1, ${this.#hiddenSize}], no ${JSON.stringify(cell.shape)}`,
            );
        }
        const inputGate = sigmoidTensor(
            add(add(matmul(input, this.#iInput), matmul(hidden, this.#iHidden)), this.#iBias),
        );
        const forgetGate = sigmoidTensor(
            add(add(matmul(input, this.#fInput), matmul(hidden, this.#fHidden)), this.#fBias),
        );
        const outputGate = sigmoidTensor(
            add(add(matmul(input, this.#oInput), matmul(hidden, this.#oHidden)), this.#oBias),
        );
        const candidate = tanh(
            add(add(matmul(input, this.#gInput), matmul(hidden, this.#gHidden)), this.#gBias),
        );
        const newCell = add(mul(forgetGate, cell), mul(inputGate, candidate));
        return {
            hidden: mul(outputGate, tanh(newCell)),
            cell: newCell,
        };
    }

    parameters(): readonly Tensor[] {
        return [
            this.#iInput,
            this.#iHidden,
            this.#iBias,
            this.#fInput,
            this.#fHidden,
            this.#fBias,
            this.#oInput,
            this.#oHidden,
            this.#oBias,
            this.#gInput,
            this.#gHidden,
            this.#gBias,
        ];
    }
}

export interface MLPOptions {
    seed?: number;
}

/** Perceptrón multicapa denso: Linear + ReLU en las capas ocultas, Linear al final. */
export class MLP implements Model {
    readonly #layers: Linear[];

    constructor(sizes: readonly number[], options: MLPOptions = {}) {
        if (sizes.length < 2) {
            throw new Error("un MLP necesita al menos una entrada y una salida");
        }
        for (const size of sizes) {
            if (!Number.isInteger(size) || size <= 0) {
                throw new Error(`tamaño de capa inválido: ${size}`);
            }
        }
        const seed = options.seed ?? 0;
        this.#layers = [];
        for (let index = 0; index < sizes.length - 1; index++) {
            this.#layers.push(
                new Linear(sizes[index] as number, sizes[index + 1] as number, { seed: seed + index }),
            );
        }
    }

    forward(input: Tensor): Tensor {
        let value = input;
        for (let index = 0; index < this.#layers.length; index++) {
            value = (this.#layers[index] as Linear).forward(value);
            if (index < this.#layers.length - 1) value = relu(value);
        }
        return value;
    }

    parameters(): readonly Tensor[] {
        return this.#layers.flatMap((layer) => layer.parameters());
    }
}

import { Tensor } from "../core/tensor.ts";
import { concatColumns, sliceColumns } from "../core/tensor-ops.ts";
import { scaledDotProductAttention } from "./attention.ts";
import { appendRows } from "./kv-cache.ts";
import type { AttentionCache } from "./kv-cache.ts";
import { Linear } from "./linear.ts";
import { rope } from "./rope.ts";
import type { RandomNumberGenerator } from "../core/random/index.ts";

export interface MultiHeadAttentionOptions {
    seed?: number;
    rng?: RandomNumberGenerator;
    rope?: boolean;
}

export interface MultiHeadAttentionResult {
    output: Tensor;
    weights: Tensor[];
}

/** Atención multi-cabeza: proyecta Q/K/V, parte en cabezas, atiende y vuelve a juntar. */
export class MultiHeadAttention {
    readonly #query: Linear;
    readonly #key: Linear;
    readonly #value: Linear;
    readonly #output: Linear;
    readonly #heads: number;
    readonly #headSize: number;
    readonly #modelSize: number;
    readonly #rope: boolean;

    constructor(modelSize: number, heads: number, options: MultiHeadAttentionOptions = {}) {
        if (!Number.isInteger(modelSize) || modelSize <= 0) {
            throw new Error(`la dimensión del modelo debe ser positiva, no ${modelSize}`);
        }
        if (!Number.isInteger(heads) || heads <= 0 || modelSize % heads !== 0) {
            throw new Error(`las cabezas deben dividir ${modelSize}, no ${heads}`);
        }
        const seed = options.seed ?? 0;
        this.#modelSize = modelSize;
        this.#heads = heads;
        this.#headSize = modelSize / heads;
        this.#rope = options.rope ?? false;
        this.#query = new Linear(modelSize, modelSize, { seed });
        this.#key = new Linear(modelSize, modelSize, { seed: seed + 1 });
        this.#value = new Linear(modelSize, modelSize, { seed: seed + 2 });
        this.#output = new Linear(modelSize, modelSize, { seed: seed + 3 });
    }

    get heads(): number {
        return this.#heads;
    }

    get headSize(): number {
        return this.#headSize;
    }

    forward(input: Tensor, options: { causal?: boolean } = {}): MultiHeadAttentionResult {
        if (input.rank !== 2 || (input.shape[1] as number) !== this.#modelSize) {
            throw new Error(
                `la entrada debe ser [secuencia, ${this.#modelSize}], no ${JSON.stringify(input.shape)}`,
            );
        }
        const queries = this.#query.forward(input);
        const keys = this.#key.forward(input);
        const values = this.#value.forward(input);
        const outputs: Tensor[] = [];
        const weights: Tensor[] = [];
        for (let head = 0; head < this.#heads; head++) {
            const start = head * this.#headSize;
            const end = start + this.#headSize;
            let queriesHead = sliceColumns(queries, start, end);
            let keysHead = sliceColumns(keys, start, end);
            if (this.#rope) {
                queriesHead = rope(queriesHead);
                keysHead = rope(keysHead);
            }
            const result = scaledDotProductAttention(
                queriesHead,
                keysHead,
                sliceColumns(values, start, end),
                { causal: options.causal ?? true },
            );
            outputs.push(result.output);
            weights.push(result.weights);
        }
        return {
            output: this.#output.forward(concatColumns(outputs)),
            weights,
        };
    }

    /** Un token nuevo contra la caché de K/V; el pasado no se recalcula. */
    forwardStep(input: Tensor, cache: AttentionCache, position: number): Tensor {
        if (input.rank !== 2 || input.shape[0] !== 1 || input.shape[1] !== this.#modelSize) {
            throw new Error(
                `la entrada debe ser [1, ${this.#modelSize}], no ${JSON.stringify(input.shape)}`,
            );
        }
        if (!Number.isInteger(position) || position < 0) {
            throw new RangeError(`posición inválida: ${position}`);
        }
        if (cache.keys.length !== this.#heads) {
            throw new Error(`la caché tiene ${cache.keys.length} cabezas, se esperaban ${this.#heads}`);
        }
        for (let head = 0; head < this.#heads; head++) {
            if ((cache.keys[head]?.shape[0] as number) !== position) {
                throw new Error(`la caché no está en la posición ${position}`);
            }
        }
        const queries = this.#query.forward(input);
        const keys = this.#key.forward(input);
        const values = this.#value.forward(input);
        const outputs: Tensor[] = [];
        for (let head = 0; head < this.#heads; head++) {
            const start = head * this.#headSize;
            const end = start + this.#headSize;
            let query = sliceColumns(queries, start, end);
            let key = sliceColumns(keys, start, end);
            if (this.#rope) {
                query = rope(query, { offset: position });
                key = rope(key, { offset: position });
            }
            const previousKeys = cache.keys[head] as Tensor;
            const previousValues = cache.values[head] as Tensor;
            cache.keys[head] = appendRows(previousKeys, key);
            cache.values[head] = appendRows(previousValues, sliceColumns(values, start, end));
            const result = scaledDotProductAttention(
                query,
                cache.keys[head] as Tensor,
                cache.values[head] as Tensor,
                { causal: false },
            );
            outputs.push(result.output);
        }
        return this.#output.forward(concatColumns(outputs));
    }

    parameters(): readonly Tensor[] {
        return [
            ...this.#query.parameters(),
            ...this.#key.parameters(),
            ...this.#value.parameters(),
            ...this.#output.parameters(),
        ];
    }
}

import { Tensor } from "../core/tensor.ts";

export interface QuantizationOptions {
    perChannel?: boolean;
}

export interface QuantizedTensor {
    data: Int8Array;
    scales: Float32Array;
    shape: number[];
    perChannel: boolean;
}

/** Cuantización simétrica a INT8: q = round(x / escala), con escala = max|x| / 127. */
export function quantizeInt8(
    tensor: Tensor,
    options: QuantizationOptions = {},
): QuantizedTensor {
    if (tensor.rank !== 1 && tensor.rank !== 2) {
        throw new Error("la cuantización espera un vector o una matriz");
    }
    const perChannel = (options.perChannel ?? true) && tensor.rank === 2;
    const columns = tensor.rank === 2 ? (tensor.shape[1] as number) : 1;
    const rows = tensor.rank === 2 ? (tensor.shape[0] as number) : tensor.size;
    const scales = new Float32Array(perChannel ? columns : 1);
    const data = new Int8Array(tensor.size);

    for (let column = 0; column < (perChannel ? columns : 1); column++) {
        let maximum = 0;
        for (let row = 0; row < rows; row++) {
            const index = tensor.rank === 2 ? row * columns + column : row;
            maximum = Math.max(maximum, Math.abs(tensor.data[index] as number));
        }
        const scale = maximum > 0 ? maximum / 127 : 1;
        scales[column] = scale;
        for (let row = 0; row < rows; row++) {
            const index = tensor.rank === 2 ? row * columns + column : row;
            const quantized = Math.round((tensor.data[index] as number) / scale);
            data[index] = Math.max(-127, Math.min(127, quantized));
        }
    }

    return { data, scales, shape: [...tensor.shape], perChannel };
}

/** Reconstruye el tensor de punto flotante desde la representación INT8. */
export function dequantizeInt8(quantized: QuantizedTensor): Tensor {
    const columns = quantized.shape.length === 2 ? (quantized.shape[1] as number) : 1;
    const rows = quantized.shape.length === 2 ? (quantized.shape[0] as number) : quantized.shape[0];
    const data = new Float32Array(quantized.data.length);
    for (let row = 0; row < (rows as number); row++) {
        for (let column = 0; column < columns; column++) {
            const index = row * columns + column;
            const scale = quantized.perChannel
                ? (quantized.scales[column] as number)
                : (quantized.scales[0] as number);
            data[index] = (quantized.data[index] as number) * scale;
        }
    }
    return new Tensor(data, quantized.shape);
}

export function quantizedBytes(quantized: QuantizedTensor): number {
    return quantized.data.byteLength + quantized.scales.byteLength;
}

export function floatBytes(tensor: Tensor): number {
    return tensor.size * Float32Array.BYTES_PER_ELEMENT;
}

import { Tensor } from "../core/tensor.ts";

export interface RMSNormOptions {
    epsilon?: number;
}

/** RMSNorm por fila: x / sqrt(mean(x²) + ε) · gamma. Sin media ni beta. */
export class RMSNorm {
    readonly gamma: Tensor;
    readonly #size: number;
    readonly #epsilon: number;

    constructor(size: number, options: RMSNormOptions = {}) {
        if (!Number.isInteger(size) || size <= 0) {
            throw new Error(`RMSNorm necesita una dimensión positiva, no ${size}`);
        }
        const epsilon = options.epsilon ?? 1e-6;
        if (!Number.isFinite(epsilon) || epsilon <= 0) {
            throw new RangeError(`epsilon debe ser positivo, no ${epsilon}`);
        }
        this.gamma = Tensor.ones([size], { requiresGrad: true });
        this.#size = size;
        this.#epsilon = epsilon;
    }

    forward(input: Tensor): Tensor {
        if (input.rank !== 2 || (input.shape[1] as number) !== this.#size) {
            throw new Error(
                `RMSNorm espera [filas, ${this.#size}], no ${JSON.stringify(input.shape)}`,
            );
        }
        const rows = input.shape[0] as number;
        const size = this.#size;
        const normalized = new Float32Array(rows * size);
        const inverse = new Float32Array(rows);
        const output = new Float32Array(rows * size);
        for (let row = 0; row < rows; row++) {
            let squares = 0;
            for (let column = 0; column < size; column++) {
                const value = input.data[row * size + column] as number;
                squares += value * value;
            }
            const rms = Math.sqrt(squares / size + this.#epsilon);
            const scale = 1 / rms;
            inverse[row] = scale;
            for (let column = 0; column < size; column++) {
                const value = (input.data[row * size + column] as number) * scale;
                normalized[row * size + column] = value;
                output[row * size + column] = value * (this.gamma.data[column] as number);
            }
        }

        const result = new Tensor(output, [rows, size]);
        if (input.requiresGrad || this.gamma.requiresGrad) {
            result.requiresGrad = true;
            result.setGraph([input, this.gamma], (gradient) => {
                const dGamma = new Float32Array(size);
                const dNormalized = new Float32Array(rows * size);
                const dInput = new Float32Array(rows * size);
                for (let row = 0; row < rows; row++) {
                    let dot = 0;
                    for (let column = 0; column < size; column++) {
                        const value = gradient.data[row * size + column] as number;
                        dGamma[column] = (dGamma[column] as number) +
                            value * (normalized[row * size + column] as number);
                        const normalizedGradient = value * (this.gamma.data[column] as number);
                        dNormalized[row * size + column] = normalizedGradient;
                        dot += normalizedGradient * (input.data[row * size + column] as number);
                    }
                    const scale = inverse[row] as number;
                    for (let column = 0; column < size; column++) {
                        dInput[row * size + column] =
                            scale * (dNormalized[row * size + column] as number) -
                            (input.data[row * size + column] as number) * dot * scale ** 3 / size;
                    }
                }
                input.addGrad(new Tensor(dInput, input.shape));
                this.gamma.addGrad(new Tensor(dGamma, [size]));
            });
        }
        return result;
    }

    parameters(): readonly Tensor[] {
        return [this.gamma];
    }
}

import { createRng } from "../core/random/index.ts";

}

function randomMatrix(
    rows: number,
    columns: number,
    rng: RandomNumberGenerator,
    limit: number,
): Tensor {

    for (let index = 0; index < data.length; index++) {
        data[index] = (rng.next() * 2 - 1) * limit;
    }

/** RNN de Elman: h_t = tanh(x_t W_x + h_{t-1} W_h + b). */
export class RNN implements RecurrentCell {
    readonly #inputWeights: Tensor;
    readonly #hiddenWeights: Tensor;
    readonly #bias: Tensor;
    readonly #inputSize: number;
    readonly #hiddenSize: number;

    constructor(inputSize: number, hiddenSize: number, options: RNNOptions = {}) {
        if (!Number.isInteger(inputSize) || inputSize <= 0) {
            throw new Error(`la entrada debe ser positiva, no ${inputSize}`);
        }
        if (!Number.isInteger(hiddenSize) || hiddenSize <= 0) {
            throw new Error(`el estado oculto debe ser positivo, no ${hiddenSize}`);
        }
        const rng = options.rng ?? createRng(options.seed ?? 0);
        this.#inputWeights = randomMatrix(
            inputSize,
            hiddenSize,
            rng,
            Math.sqrt(6 / (inputSize + hiddenSize)),
        );
        this.#hiddenWeights = randomMatrix(
            hiddenSize,
            hiddenSize,
            rng,
            Math.sqrt(6 / (hiddenSize + hiddenSize)),
        );
        this.#bias = Tensor.zeros([hiddenSize], { requiresGrad: true });
        this.#inputSize = inputSize;
        this.#hiddenSize = hiddenSize;
    }

    get inputSize(): number {
        return this.#inputSize;
    }

    get hiddenSize(): number {
        return this.#hiddenSize;
    }

    initialHidden(): Tensor {
        return Tensor.zeros([1, this.#hiddenSize]);
    }

    initial(): RecurrentState {
        return { hidden: this.initialHidden() };
    }

    advance(input: Tensor, state: RecurrentState): RecurrentState {
        return { hidden: this.step(input, state.hidden) };
    }

    step(input: Tensor, hidden: Tensor): Tensor {
        if (input.rank !== 2 || input.shape[0] !== 1 || input.shape[1] !== this.#inputSize) {
            throw new Error(
                `la entrada debe ser [1, ${this.#inputSize}], no ${JSON.stringify(input.shape)}`,
            );
        }
        if (hidden.rank !== 2 || hidden.shape[0] !== 1 || hidden.shape[1] !== this.#hiddenSize) {
            throw new Error(
                `el estado debe ser [1, ${this.#hiddenSize}], no ${JSON.stringify(hidden.shape)}`,
            );
        }
        const inputPart = matmul(input, this.#inputWeights);
        const hiddenPart = add(matmul(hidden, this.#hiddenWeights), this.#bias);
        return tanh(add(inputPart, hiddenPart));
    }

    forward(inputs: readonly Tensor[]): Tensor[] {
        const states: Tensor[] = [];
        let hidden = this.initialHidden();
        for (const input of inputs) {
            hidden = this.step(input, hidden);
            states.push(hidden);
        }
        return states;
    }

    parameters(): readonly Tensor[] {
        return [this.#inputWeights, this.#hiddenWeights, this.#bias];
    }
}


// Teoría: docs/teoria/secuencias/rope.md

import { Tensor } from "../core/tensor.ts";

export interface RoPEOptions {
    base?: number;
    offset?: number;
}

/**
 * Rotary Position Embedding: rota cada par de dimensiones por `posición · θ_i`.
 * La norma se conserva y el producto q·k depende de la distancia entre posiciones.
 */
export function rope(tensor: Tensor, options: RoPEOptions = {}): Tensor {
    if (tensor.rank !== 2) throw new Error("RoPE espera una matriz [secuencia, dimensión]");
    const sequence = tensor.shape[0] as number;
    const dimension = tensor.shape[1] as number;
    if (dimension % 2 !== 0) {
        throw new Error(`RoPE necesita una dimensión par, no ${dimension}`);
    }
    const base = options.base ?? 10000;
    if (!Number.isFinite(base) || base <= 0) {
        throw new RangeError(`la base de RoPE debe ser positiva, no ${base}`);
    }
    const offset = options.offset ?? 0;
    if (!Number.isInteger(offset) || offset < 0) {
        throw new RangeError(`el desplazamiento debe ser un entero no negativo, no ${offset}`);
    }

    const half = dimension / 2;
    const cosines = new Float32Array(sequence * half);
    const sines = new Float32Array(sequence * half);
    const output = new Float32Array(sequence * dimension);
    for (let row = 0; row < sequence; row++) {
        const position = offset + row;
        for (let pair = 0; pair < half; pair++) {
            const angle = position * base ** (-2 * pair / dimension);
            const cosine = Math.cos(angle);
            const sine = Math.sin(angle);
            cosines[row * half + pair] = cosine;
            sines[row * half + pair] = sine;
            const first = tensor.data[row * dimension + pair * 2] as number;
            const second = tensor.data[row * dimension + pair * 2 + 1] as number;
            output[row * dimension + pair * 2] = first * cosine - second * sine;
            output[row * dimension + pair * 2 + 1] = first * sine + second * cosine;
        }
    }

    const result = new Tensor(output, [sequence, dimension]);
    if (tensor.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([tensor], (gradient) => {
            const values = new Float32Array(sequence * dimension);
            for (let row = 0; row < sequence; row++) {
                for (let pair = 0; pair < half; pair++) {
                    const cosine = cosines[row * half + pair] as number;
                    const sine = sines[row * half + pair] as number;
                    const first = gradient.data[row * dimension + pair * 2] as number;
                    const second = gradient.data[row * dimension + pair * 2 + 1] as number;
                    values[row * dimension + pair * 2] = first * cosine + second * sine;
                    values[row * dimension + pair * 2 + 1] = -first * sine + second * cosine;
                }
            }
            tensor.addGrad(new Tensor(values, tensor.shape));
        });
    }
    return result;
}

import type { RandomNumberGenerator } from "../core/random/index.ts";

export interface SamplingOptions {
    temperature?: number;
    topK?: number;
    topP?: number;
}

function checked(options: number | SamplingOptions): Required<Pick<SamplingOptions, "temperature" | "topP">> & {
    topK?: number;
} {
    const config = typeof options === "number" ? { temperature: options } : options;
    const temperature = config.temperature ?? 1;
    if (!Number.isFinite(temperature) || temperature <= 0) {
        throw new RangeError(`la temperatura debe ser positiva, no ${temperature}`);
    }
    if (
        config.topK !== undefined &&
        (!Number.isInteger(config.topK) || config.topK <= 0)
    ) {
        throw new RangeError(`top-k debe ser un entero positivo, no ${config.topK}`);
    }
    const topP = config.topP ?? 1;
    if (!Number.isFinite(topP) || topP <= 0 || topP > 1) {
        throw new RangeError(`top-p debe estar en (0, 1], no ${topP}`);
    }
    return { temperature, topP, ...(config.topK === undefined ? {} : { topK: config.topK }) };
}

/** Muestrea un índice de una fila de logits con temperatura, top-k y top-p. */
export function sampleFromValues(
    values: ArrayLike<number>,
    options: number | SamplingOptions,
    rng: RandomNumberGenerator,
): number {
    const { temperature, topK, topP } = checked(options);
    if (values.length === 0) throw new Error("no se puede muestrear una fila vacía");

    const candidates = Array.from({ length: values.length }, (_, index) => ({
        index,
        logit: values[index] as number,
    })).sort((first, second) => second.logit - first.logit);
    const kept = topK === undefined ? candidates : candidates.slice(0, topK);
    const maximum = (kept[0] as { logit: number }).logit;
    const probabilities = kept.map((candidate) =>
        Math.exp((candidate.logit - maximum) / temperature),
    );
    let total = probabilities.reduce((sum, value) => sum + value, 0);

    if (topP < 1) {
        let cumulative = 0;
        let cutoff = probabilities.length;
        for (let index = 0; index < probabilities.length && cumulative < topP; index++) {
            cumulative += (probabilities[index] as number) / total;
            cutoff = index + 1;
        }
        probabilities.length = cutoff;
        total = probabilities.reduce((sum, value) => sum + value, 0);
    }

    let threshold = rng.next() * total;
    for (let index = 0; index < probabilities.length; index++) {
        threshold -= probabilities[index] as number;
        if (threshold <= 0) return (kept[index] as { index: number }).index;
    }
    return (kept[probabilities.length - 1] as { index: number }).index;
}

import { Tensor } from "../core/tensor.ts";
import { elementwiseRaw, unaryRaw } from "../core/tensor-ops.ts";

/** σ(x) = 1 / (1 + e^{-x}), evaluada sin desbordar el exponente. */
export function sigmoid(x: number): number {
    const e = Math.exp(-Math.abs(x));
    return x >= 0 ? 1 / (1 + e) : e / (1 + e);
}

/** σ'(x) = σ(x) · (1 − σ(x)). */
export function sigmoidDerivative(x: number): number {
    const value = sigmoid(x);
    return value * (1 - value);
}

/** σ elemento a elemento sobre un tensor; registra gradiente si hace falta. */
export function sigmoidTensor(tensor: Tensor): Tensor {
    const result = unaryRaw(tensor, sigmoid);
    if (tensor.requiresGrad) {
        result.requiresGrad = true;
        result.setGraph([tensor], (gradient) => {
            const derivative = unaryRaw(tensor, sigmoidDerivative);
            tensor.addGrad(elementwiseRaw(gradient, derivative, (left, right) => left * right));
        });
    }
    return result;
}

import { Tensor } from "../core/tensor.ts";
import { add } from "../core/tensor-ops.ts";
import { FeedForward } from "./feedforward.ts";
import type { AttentionCache } from "./kv-cache.ts";
import { LayerNorm } from "./layernorm.ts";
import { MultiHeadAttention } from "./multihead.ts";
import { RMSNorm } from "./rmsnorm.ts";
import { SwiGLU } from "./swiglu.ts";

export interface TransformerBlockOptions {
    hiddenSize?: number;
    seed?: number;
    rope?: boolean;
    attention?: boolean;
    ffn?: boolean;
    normKind?: "layer" | "rms";
    ffnKind?: "relu" | "swiglu";
    parallel?: boolean;
}

export interface TransformerBlockResult {
    output: Tensor;
    weights: Tensor[];
}

/**
 * Bloque transformer pre-norm:
 *   x = x + MultiHead(LayerNorm(x))
 *   y = x + FFN(LayerNorm(x))
 * Las piezas se pueden apagar para la ablación.
 */
export class TransformerBlock {
    readonly #norm1: LayerNorm | RMSNorm;
    readonly #attention: MultiHeadAttention | null;
    readonly #norm2: LayerNorm | RMSNorm;
    readonly #ffn: FeedForward | SwiGLU | null;
    readonly #parallel: boolean;

    constructor(modelSize: number, heads: number, options: TransformerBlockOptions = {}) {
        const hiddenSize = options.hiddenSize ?? modelSize * 4;
        const seed = options.seed ?? 0;
        const normKind = options.normKind ?? "layer";
        const ffnKind = options.ffnKind ?? "relu";
        this.#parallel = options.parallel ?? false;
        this.#norm1 = normKind === "rms" ? new RMSNorm(modelSize) : new LayerNorm(modelSize);
        this.#attention =
            options.attention ?? true
                ? new MultiHeadAttention(modelSize, heads, {
                      seed,
                      rope: options.rope ?? false,
                  })
                : null;
        this.#norm2 = normKind === "rms" ? new RMSNorm(modelSize) : new LayerNorm(modelSize);
        this.#ffn =
            options.ffn ?? true
                ? ffnKind === "swiglu"
                    ? new SwiGLU(modelSize, hiddenSize, { seed: seed + 4 })
                    : new FeedForward(modelSize, hiddenSize, { seed: seed + 4 })
                : null;
    }

    forward(input: Tensor, options: { causal?: boolean } = {}): TransformerBlockResult {
        if (this.#parallel) {
            let output = input;
            let weights: Tensor[] = [];
            if (this.#attention) {
                const attention = this.#attention.forward(this.#norm1.forward(input), {
                    causal: options.causal ?? true,
                });
                output = add(output, attention.output);
                weights = attention.weights;
            }
            if (this.#ffn) output = add(output, this.#ffn.forward(this.#norm2.forward(input)));
            return { output, weights };
        }
        let first = input;
        let weights: Tensor[] = [];
        if (this.#attention) {
            const attention = this.#attention.forward(this.#norm1.forward(input), {
                causal: options.causal ?? true,
            });
            first = add(input, attention.output);
            weights = attention.weights;
        }
        const output = this.#ffn
            ? add(first, this.#ffn.forward(this.#norm2.forward(first)))
            : first;
        return { output, weights };
    }

    /** Un token nuevo con caché de atención; requiere que la atención esté activa. */
    forwardCached(input: Tensor, cache: AttentionCache, position: number): Tensor {
        if (!this.#attention) throw new Error("el bloque no tiene atención");
        if (this.#parallel) {
            let output = add(
                input,
                this.#attention.forwardStep(this.#norm1.forward(input), cache, position),
            );
            if (this.#ffn) output = add(output, this.#ffn.forward(this.#norm2.forward(input)));
            return output;
        }
        const first = add(
            input,
            this.#attention.forwardStep(this.#norm1.forward(input), cache, position),
        );
        return this.#ffn
            ? add(first, this.#ffn.forward(this.#norm2.forward(first)))
            : first;
    }

    parameters(): readonly Tensor[] {
        const parameters: Tensor[] = [];
        if (this.#attention) {
            parameters.push(...this.#norm1.parameters(), ...this.#attention.parameters());
        }
        if (this.#ffn) {
            parameters.push(...this.#norm2.parameters(), ...this.#ffn.parameters());
        }
        return parameters;
    }
}

import type { Tensor } from "../core/tensor.ts";

/** Recorta la norma global de los gradientes; devuelve la norma antes de recortar. */
export function clipGradients(parameters: readonly Tensor[], maxNorm: number): number {
    if (!Number.isFinite(maxNorm) || maxNorm <= 0) {
        throw new RangeError(`la norma máxima debe ser positiva, no ${maxNorm}`);
    }
    let squares = 0;
    for (const parameter of parameters) {
        const gradient = parameter.grad;
        if (gradient) {
            for (const value of gradient.data) squares += value * value;
        }
    }
    const norm = Math.sqrt(squares);
    if (norm > maxNorm) {
        const scale = maxNorm / norm;
        for (const parameter of parameters) {
            const gradient = parameter.grad;
            if (gradient) {
                for (let index = 0; index < gradient.data.length; index++) {
                    gradient.data[index] = (gradient.data[index] as number) * scale;
                }
            }
        }
    }
    return norm;
}


// Teoría: docs/teoria/entrenamiento/sgd.md

import { Tensor } from "../core/tensor.ts";

/** Contrato de un optimizador: actualiza los parámetros y limpia sus gradientes. */
export interface Optimizer {
    step(): void;
    zeroGrad(): void;
}

/** Descenso de gradiente estocástico: θ ← θ − η · ∇L. */
export class SGD implements Optimizer {
    readonly #parameters: readonly Tensor[];
    #learningRate: number;

    constructor(parameters: readonly Tensor[], learningRate: number) {
        if (!Number.isFinite(learningRate) || learningRate <= 0) {
            throw new RangeError(`la tasa de aprendizaje debe ser un número positivo, no ${learningRate}`);
        }
        this.#parameters = parameters;
        this.#learningRate = learningRate;
    }

    get learningRate(): number {
        return this.#learningRate;
    }

    set learningRate(value: number) {
        if (!Number.isFinite(value) || value <= 0) {
            throw new RangeError(`la tasa de aprendizaje debe ser un número positivo, no ${value}`);
        }
        this.#learningRate = value;
    }

    step(): void {
        for (const parameter of this.#parameters) {
            const gradient = parameter.grad;
            if (parameter.requiresGrad && gradient !== null) {
                for (let index = 0; index < parameter.size; index++) {
                    parameter.data[index] =
                        (parameter.data[index] as number) -
                        this.#learningRate * (gradient.data[index] as number);
                }
            }
        }
    }

    zeroGrad(): void {
        for (const parameter of this.#parameters) parameter.zeroGrad();
    }
}

import type { Tensor } from "../core/tensor.ts";

export interface ConfusionMatrix {
    truePositive: number;
    falsePositive: number;
    trueNegative: number;
    falseNegative: number;
    ties: number;
    total: number;
}

export interface DirectionReport {
    matrix: ConfusionMatrix;
    evaluated: number;
    coverage: number;
    accuracy: number;
    precision: number;
    recall: number;
    specificity: number;
    f1: number;
    balancedAccuracy: number;
    mcc: number;
    predictedUpRate: number;
    actualUpRate: number;
}

export interface DirectionalErrors {
    up: number;
    down: number;
    upCount: number;
    downCount: number;
}

function checkedSize(prediction: Tensor, target: Tensor): number {
    const sameShape =
        prediction.shape.length === target.shape.length &&
        prediction.shape.every((dimension, axis) => dimension === target.shape[axis]);
    if (!sameShape) {
        throw new Error(
            `formas distintas: ${JSON.stringify(prediction.shape)} y ${JSON.stringify(target.shape)}`,
        );
    }
    if (prediction.size === 0) throw new Error("no se puede medir una predicción vacía");
    return prediction.size;
}

/**
 * Matriz de confusión de la dirección. La clase positiva es «sube» (retorno > 0); los objetivos
 * exactamente 0 se cuentan como `ties` y quedan fuera de la matriz. Una predicción 0 cuenta como
 * «baja».
 */
export function confusionMatrix(prediction: Tensor, target: Tensor): ConfusionMatrix {
    const size = checkedSize(prediction, target);
    let truePositive = 0;
    let falsePositive = 0;
    let trueNegative = 0;
    let falseNegative = 0;
    let ties = 0;
    for (let index = 0; index < size; index++) {
        const actual = target.data[index] as number;
        const predicted = prediction.data[index] as number;
        if (actual === 0) {
            ties += 1;
        } else {
            const actualUp = actual > 0;
            const predictedUp = predicted > 0;
            if (actualUp && predictedUp) truePositive += 1;
            else if (actualUp) falseNegative += 1;
            else if (predictedUp) falsePositive += 1;
            else trueNegative += 1;
        }
    }
    return { truePositive, falsePositive, trueNegative, falseNegative, ties, total: size };
}

export function directionReport(prediction: Tensor, target: Tensor): DirectionReport {
    const matrix = confusionMatrix(prediction, target);
    const evaluated = matrix.total - matrix.ties;
    const divide = (numerator: number, denominator: number): number =>
        denominator > 0 ? numerator / denominator : 0;

    const accuracy = divide(matrix.truePositive + matrix.trueNegative, evaluated);
    const precision = divide(matrix.truePositive, matrix.truePositive + matrix.falsePositive);
    const recall = divide(matrix.truePositive, matrix.truePositive + matrix.falseNegative);
    const specificity = divide(matrix.trueNegative, matrix.trueNegative + matrix.falsePositive);
    const f1 = precision + recall > 0 ? (2 * precision * recall) / (precision + recall) : 0;
    const balancedAccuracy = (recall + specificity) / 2;
    const mccDenominator = Math.sqrt(
        (matrix.truePositive + matrix.falsePositive) *
            (matrix.truePositive + matrix.falseNegative) *
            (matrix.trueNegative + matrix.falsePositive) *
            (matrix.trueNegative + matrix.falseNegative),
    );
    const mcc =
        mccDenominator > 0
            ? (matrix.truePositive * matrix.trueNegative -
                  matrix.falsePositive * matrix.falseNegative) /
              mccDenominator
            : 0;

    return {
        matrix,
        evaluated,
        coverage: divide(evaluated, matrix.total),
        accuracy,
        precision,
        recall,
        specificity,
        f1,
        balancedAccuracy,
        mcc,
        predictedUpRate: divide(matrix.truePositive + matrix.falsePositive, evaluated),
        actualUpRate: divide(matrix.truePositive + matrix.falseNegative, evaluated),
    };
}

/** MAE separado por la dirección real; los ceros quedan fuera. */
export function errorsByDirection(prediction: Tensor, target: Tensor): DirectionalErrors {
    const size = checkedSize(prediction, target);
    let upError = 0;
    let downError = 0;
    let upCount = 0;
    let downCount = 0;
    for (let index = 0; index < size; index++) {
        const actual = target.data[index] as number;
        if (actual !== 0) {
            const error = Math.abs((prediction.data[index] as number) - actual);
            if (actual > 0) {
                upError += error;
                upCount += 1;
            } else {
                downError += error;
                downCount += 1;
            }
        }
    }
    return {
        up: upCount > 0 ? upError / upCount : 0,
        down: downCount > 0 ? downError / downCount : 0,
        upCount,
        downCount,
    };
}

function checkedSize(prediction: Tensor, target: Tensor): number {
    const sameShape =
        prediction.shape.length === target.shape.length &&
        prediction.shape.every((dimension, axis) => dimension === target.shape[axis]);
    if (!sameShape) {
        throw new Error(
            `formas distintas: ${JSON.stringify(prediction.shape)} y ${JSON.stringify(target.shape)}`,
        );
    }
    if (prediction.size === 0) throw new Error("no se puede medir una predicción vacía");
    return prediction.size;
}

export function meanAbsoluteError(prediction: Tensor, target: Tensor): number {
    const size = checkedSize(prediction, target);
    let total = 0;
    for (let index = 0; index < size; index++) {
        total += Math.abs((prediction.data[index] as number) - (target.data[index] as number));
    }
    return total / size;
}

export function rootMeanSquaredError(prediction: Tensor, target: Tensor): number {
    const size = checkedSize(prediction, target);
    let total = 0;
    for (let index = 0; index < size; index++) {
        const difference = (prediction.data[index] as number) - (target.data[index] as number);
        total += difference * difference;
    }
    return Math.sqrt(total / size);
}

/** Proporción de signos que coinciden; un cero solo acierta contra otro cero. */
export function directionalAccuracy(prediction: Tensor, target: Tensor): number {
    const size = checkedSize(prediction, target);
    let correct = 0;
    for (let index = 0; index < size; index++) {
        if (
            Math.sign(prediction.data[index] as number) === Math.sign(target.data[index] as number)
        ) {
            correct += 1;
        }
    }
    return correct / size;
}

/** Sesgo: media de (predicción − objetivo); positivo = optimista. */
export function meanError(prediction: Tensor, target: Tensor): number {
    const size = checkedSize(prediction, target);
    let total = 0;
    for (let index = 0; index < size; index++) {
        total += (prediction.data[index] as number) - (target.data[index] as number);
    }
    return total / size;
}

/** Coeficiente de determinación R²; 0 si el objetivo no tiene varianza. */
export function coefficientOfDetermination(prediction: Tensor, target: Tensor): number {
    const size = checkedSize(prediction, target);
    let mean = 0;
    for (let index = 0; index < size; index++) mean += target.data[index] as number;
    mean /= size;
    let residual = 0;
    let total = 0;
    for (let index = 0; index < size; index++) {
        const difference = (prediction.data[index] as number) - (target.data[index] as number);
        residual += difference * difference;
        const centered = (target.data[index] as number) - mean;
        total += centered * centered;
    }
    return total > 0 ? 1 - residual / total : 0;
}

/** Correlación de Pearson entre predicción y objetivo; 0 si alguna no tiene varianza. */
export function correlation(prediction: Tensor, target: Tensor): number {
    const size = checkedSize(prediction, target);
    let meanPrediction = 0;
    let meanTarget = 0;
    for (let index = 0; index < size; index++) {
        meanPrediction += prediction.data[index] as number;
        meanTarget += target.data[index] as number;
    }
    meanPrediction /= size;
    meanTarget /= size;
    let covariance = 0;
    let variancePrediction = 0;
    let varianceTarget = 0;
    for (let index = 0; index < size; index++) {
        const predictionDelta = (prediction.data[index] as number) - meanPrediction;
        const targetDelta = (target.data[index] as number) - meanTarget;
        covariance += predictionDelta * targetDelta;
        variancePrediction += predictionDelta * predictionDelta;
        varianceTarget += targetDelta * targetDelta;
    }
    const denominator = Math.sqrt(variancePrediction * varianceTarget);
    return denominator > 0 ? covariance / denominator : 0;
}

/**
 * Set de regresión sintético: un activo con GBM, un segundo activo correlacionado y una volatilidad
 * móvil. El objetivo es el retorno del día siguiente en porcentaje.
 */
export function syntheticAssetSet(options: SyntheticAssetOptions = {}): RegressionSet {
    const days = options.days ?? 1200;
    const seed = options.seed ?? 1;
    if (!Number.isInteger(days) || days < WINDOW + 10) {
        throw new Error(`hacen falta al menos ${WINDOW + 10} días, no ${days}`);
    }

    const rng = createRng(seed);
    const normal = (): number => {
        const first = 1 - rng.next();
        const second = rng.next();
        return Math.sqrt(-2 * Math.log(first)) * Math.cos(2 * Math.PI * second);
    };

    const returns1 = new Float32Array(days);
    const returns2 = new Float32Array(days);
    for (let day = 0; day < days; day++) {
        const common = normal();
        returns1[day] = 0.0003 + 0.01 * (0.6 * common + 0.8 * normal());
        returns2[day] = 0.0002 + 0.012 * (0.5 * common + 0.866 * normal());
    }

    const rows = days - WINDOW - 1;
    const features = new Float32Array(rows * FEATURES);
    const targets = new Float32Array(rows);
    for (let row = 0; row < rows; row++) {
        const day = row + WINDOW;
        let sum = 0;
        let squares = 0;
        for (let index = day - WINDOW + 1; index <= day; index++) {
            const value = returns1[index] as number;
            sum += value;
            squares += value * value;
        }
        const mean = sum / WINDOW;
        const variance = Math.max(squares / WINDOW - mean * mean, 0);
        features[row * FEATURES] = returns1[day] as number;
        features[row * FEATURES + 1] = Math.sqrt(variance);
        features[row * FEATURES + 2] = returns2[day] as number;
        targets[row] = (returns1[day + 1] as number) * 100;
    }

    const trainEnd = Math.floor(rows * TRAIN_FRACTION);
    const validationEnd = Math.floor(rows * VALIDATION_FRACTION);

    const means = [0, 0, 0];
    for (let row = 0; row < trainEnd; row++) {
        for (let feature = 0; feature < FEATURES; feature++) {
            means[feature] =
                (means[feature] as number) + (features[row * FEATURES + feature] as number);
        }
    }
    for (let feature = 0; feature < FEATURES; feature++) {
        means[feature] = (means[feature] as number) / trainEnd;
    }

    const deviations = [0, 0, 0];
    for (let row = 0; row < trainEnd; row++) {
        for (let feature = 0; feature < FEATURES; feature++) {
            const delta =
                (features[row * FEATURES + feature] as number) - (means[feature] as number);
            deviations[feature] = (deviations[feature] as number) + delta * delta;
        }
    }
    for (let feature = 0; feature < FEATURES; feature++) {
        const deviation = Math.sqrt((deviations[feature] as number) / trainEnd);
        deviations[feature] = deviation > 0 ? deviation : 1;
    }

    const normalized = new Float32Array(rows * FEATURES);
    for (let row = 0; row < rows; row++) {
        for (let feature = 0; feature < FEATURES; feature++) {
            normalized[row * FEATURES + feature] =
                ((features[row * FEATURES + feature] as number) - (means[feature] as number)) /
                (deviations[feature] as number);
        }
    }

    const split = (start: number, end: number): Split => ({
        input: new Tensor(normalized.slice(start * FEATURES, end * FEATURES), [
            end - start,
            FEATURES,
        ]),
        target: new Tensor(targets.slice(start, end), [end - start, 1]),
    });
    const train = split(0, trainEnd);
    const validation = split(trainEnd, validationEnd);
    const test = split(validationEnd, rows);

    let baseline = 0;
    for (const value of test.target.data) baseline += Math.abs(value);
    baseline /= test.target.size;

    return {
        name: "activo-sintetico",
        description: `GBM de ${days} días (semilla ${seed}); retorno del día siguiente en %`,
        kind: "regression",
        metricName: "MAE",
        train,
        validation,
        test,
        loss: (prediction, part) => mse(prediction, part.target),
        metric: (prediction, part) => meanAbsoluteError(prediction, part.target),
        baseline,
    };
}

import type { Tensor } from "../core/tensor.ts";
import type { Optimizer } from "../optim/sgd.ts";

export interface TrainOptions {
    steps: number;
    optimizer: Optimizer;
    /** Pérdida para la predicción del paso; debe devolver un escalar. */
    loss: (prediction: Tensor) => Tensor;
    onStep?: (step: number, loss: number) => void;
}

/**
 * Bucle de entrenamiento: zeroGrad → forward → pérdida → backward → step.
 * Devuelve la pérdida de cada paso, en orden.
 */
export function train(
    forward: (input: Tensor) => Tensor,
    input: Tensor,
    options: TrainOptions,
): number[] {
    if (!Number.isInteger(options.steps) || options.steps < 0) {
        throw new Error(`el número de pasos debe ser un entero ≥ 0, no ${options.steps}`);
    }
    const history: number[] = [];
    for (let step = 0; step < options.steps; step++) {
        options.optimizer.zeroGrad();
        const loss = options.loss(forward(input));
        if (loss.rank !== 0) throw new Error("la pérdida debe ser un escalar");
        loss.backward();
        options.optimizer.step();
        const value = loss.get();
        history.push(value);
        options.onStep?.(step, value);
    }
    return history;
}

const tracked = (values: number[], shape?: readonly number[]): Tensor =>
    Tensor.from(values, shape ?? [values.length], { requiresGrad: true });

const gradientOf = (tensor: Tensor): number[] => [...(tensor.grad?.data ?? [])];

function numericGradient(f: (input: Tensor) => Tensor, input: Tensor, epsilon = 1e-2): Float32Array {
    const original = input.data.slice();
    const result = new Float32Array(input.size);
    for (let index = 0; index < input.size; index++) {
        input.data[index] = (original[index] as number) + epsilon;
        const plus = f(input).data.reduce((total, value) => total + value, 0);
        input.data[index] = (original[index] as number) - epsilon;
        const minus = f(input).data.reduce((total, value) => total + value, 0);
        input.data[index] = original[index] as number;
        result[index] = (plus - minus) / (2 * epsilon);
    }
    return result;
}

function numericGradientOf(forward: () => Tensor, parameter: Tensor, epsilon = 1e-2): Float32Array {
    const original = parameter.data.slice();
    const result = new Float32Array(parameter.size);
    for (let index = 0; index < parameter.size; index++) {
        parameter.data[index] = (original[index] as number) + epsilon;
        const plus = forward().get();
        parameter.data[index] = (original[index] as number) - epsilon;
        const minus = forward().get();
        parameter.data[index] = original[index] as number;
        result[index] = (plus - minus) / (2 * epsilon);
    }
    return result;
}

test("gradcheck de un MLP de dos capas", () => {
    const input = Tensor.from([0.5, -0.3, 0.8, 0.1], [2, 2]);
    const target = Tensor.from([1, 0], [2, 1]);
    const weights1 = tracked([0.2, -0.1, 0.3, 0.15, -0.25, 0.05], [2, 3]);
    const bias1 = tracked([0.01, -0.02, 0.03], [3]);
    const weights2 = tracked([0.4, -0.3, 0.2], [3, 1]);
    const bias2 = tracked([-0.05], [1]);

    const forward = (): Tensor => {
        const hidden = sigmoidTensor(add(matmul(input, weights1), bias1));
        const prediction = sigmoidTensor(add(matmul(hidden, weights2), bias2));
        const difference = sub(prediction, target);
        return mean(mul(difference, difference));
    };

    forward().backward();

    for (const parameter of [weights1, bias1, weights2, bias2]) {
        const numeric = numericGradientOf(forward, parameter);
        for (let index = 0; index < parameter.size; index++) {
            const analytic = (parameter.grad?.data[index] ?? 0) as number;
            const expected = numeric[index] as number;
            assert.ok(Math.abs(analytic - expected) < 2e-2, `en ${index}: ${analytic} vs ${expected}`);
        }
    }
});

/** Generador de prueba: devuelve una secuencia fija de uint32 en ciclo. */
class FixedSequence extends Uint32RandomGenerator {
    #values: number[];
    #index = 0;

    constructor(values: number[]) {
        super();
        this.#values = values;
    }

    protected nextUint32(): number {
        const value = this.#values[this.#index % this.#values.length] as number;
        this.#index += 1;
        return value;
    }
}
