    const input = tracked([0, 1]);
    sum(exp(input)).backward();
    assert.ok(Math.abs((input.grad?.data[0] ?? 0) - 1) < 1e-6);
    assert.ok(Math.abs((input.grad?.data[1] ?? 0) - Math.E) < 1e-5);

    const input = tracked([1, 2, 4]);
    sum(log(input)).backward();

        assert.ok(Math.abs((input.grad?.data[index] ?? 0) - expected) < 1e-6);
    });

test("mutación, cruce uniforme y cruce aritmético", () => {
    const mutation = new GaussianMutation({ sigma: 0.5 });
    const parent = new Float32Array([1, 2]);
    const child = mutation.apply([parent], createRng(1));
    assert.equal(child.length, 2);
    assert.notDeepEqual([...child], [...parent]);
    assert.deepEqual(
        [...mutation.apply([parent], createRng(1))],
        [...mutation.apply([parent], createRng(1))],
    );

    const first = new Float32Array([1, 2, 3]);
    const second = new Float32Array([4, 5, 6]);
    const uniform = new UniformCrossover();
    const mixed = uniform.apply([first, second], createRng(2));
    mixed.forEach((value, index) => {
        assert.ok(
            value === (first[index] as number) || value === (second[index] as number),
            `gen ${index} no viene de un padre`,
        );
    });

    const arithmetic = new ArithmeticCrossover();
    const averaged = arithmetic.apply([first, second], createRng(3));
    averaged.forEach((value, index) => {
        const low = Math.min(first[index] as number, second[index] as number);
        const high = Math.max(first[index] as number, second[index] as number);
        assert.ok(value >= low && value <= high, `gen ${index} fuera de rango`);
    });

    assert.throws(() => uniform.apply([first, new Float32Array([1])], createRng(4)), Error);
});

test("el gradiente de Q, K y V coincide con diferencias finitas", () => {
    const q = Tensor.from([0.5, 0.2, -0.1, 0.7], [2, 2], { requiresGrad: true });
    const k = Tensor.from([0.3, -0.4, 0.6, 0.1], [2, 2], { requiresGrad: true });
    const v = Tensor.from([1, -2, 3, 0.5], [2, 2], { requiresGrad: true });
    const loss = () => sum(scaledDotProductAttention(q, k, v).output).get();
    sum(scaledDotProductAttention(q, k, v).output).backward();

    for (const [tensor, name] of [
        [q, "Q"],
        [k, "K"],
        [v, "V"],
    ] as const) {
        for (let index = 0; index < tensor.size; index++) {
            const original = tensor.data[index] as number;
            tensor.data[index] = original + 1e-2;
            const plus = loss();
            tensor.data[index] = original - 1e-2;
            const minus = loss();
            tensor.data[index] = original;
            const numeric = (plus - minus) / 0.02;
            const analytic = tensor.grad?.data[index] ?? 0;
            assert.ok(
                Math.abs(analytic - numeric) < 1e-2,
                `${name}[${index}]: ${analytic} vs ${numeric}`,
            );
        }
    }
});
